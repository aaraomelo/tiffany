\begin{abstract}
O \textbf{Corpo Universal} é o quadro dos operadores que a bateria já
mediu, organizado pela cadeia
\[
\boxed{
\text{corpo}\to\text{norma}\to\text{ordem}\to\text{observador}
\to\text{transformação}\to\text{dual}.
}
\]
As \textbf{cinco operações} --- Soma, Multiplicação, Divisão, Dual,
Inversão --- entram cada uma com a sua conservação \emph{medida}; o
\textbf{observador} é a escada $I_1\prec I_2\prec I_3\prec\cdots$; a
\textbf{transformada universal} é a avaliação nas folhas, realizada
inteira pela matriz companheira; e a \textbf{membrana de Dirac} é a
raiz quadrada da transição do Metrónomo: não existe no próprio andar
(o dual proíbe, $\det^2=-1$), existe \emph{inteira} um andar acima,
aterra na diagonal exatamente nas transições, e o seu espectro vive na
\textbf{borda dobrada} $x^{4}-mx^{2}-1$ --- a dobra temporal
$x\mapsto x^{2}$ a aceder representantes na dimensão acima. O
\textbf{Corpo de Peano} (\code{papers/corpo\_peano.tex}) é

\section{As cinco operações --- cada uma com a sua conservação}\label{sec:cinco}

\begin{definicao}[as cinco operações do corpo universal]\label{def:cinco}
\begin{center}\small
\begin{tabular}{@{}lll@{}}
\toprule
operação & conservação medida & medidor \\
\midrule
\textbf{Soma} $\oplus$ (a dobra/clone, Lei~4) & $E(u\oplus u)=2E(u)$; a retração devolve & \code{observador\_torre} \S T5 \\
\textbf{Multiplicação} $\otimes$ (fusão) & $E(u\otimes v)=E(u)E(v)$ --- $N(xy)=N(x)N(y)$ & \code{observador\_torre} \S T3 \\
\textbf{Divisão} (a fibra) & $x_i=z_{ij}/v_j$ exata; $E(u)=E(z)/E(v)$ & \code{observador\_torre} \S T4 \\
\textbf{Dual} $\dagger$ (a involução) & $\sigma\sigma^{\dagger}=-1$; $\lambda^{+}+\lambda^{-}=0$ & \code{lyapunov\_refletido}; \emph{Peano} \\
\textbf{Inversão} (a volta) & admissível $\iff$ tem volta $\wedge$ conserva a escada & \code{ponte\_universal\_metalica} \S P3 \\
\bottomrule
\end{tabular}
\end{center}
A Soma conserva \emph{adicionando}; a Multiplicação conserva
\emph{multiplicando} --- o par aditivo/multiplicativo da torre; a
Divisão é a \textbf{fibra inversa} da fusão (dividir custa exatamente o
$\det$: $\operatorname{adj}(M)\,M=\det(M)\,I$); o Dual guarda a segunda
metade; a Inversão é o critério de admissibilidade.
\end{definicao}

\begin{definicao}[o Inversor]\label{def:inversor}
Uma transformação $T$ é \textbf{admissível} quando
\[
\boxed{
\exists\,T^{-1}\ \text{(a volta)}
\qquad\land\qquad
I\bigl(T^{-1}(T(x))\bigr)=I(x)\ \text{na escada completa}.
}
\]
O \textbf{Inversor} é o papel funcional que decide: mede a volta contra
$(E,\Phi,\Phi_2,\ldots)$ e rejeita no primeiro desvio. Medido: a
transformada metálica passa (volta inteira nas unidades, escada
idêntica); a membrana de Dirac passa
(\code{tests/dirac\_transicao.js} \S D5).
\end{definicao}

\section{O observador --- a escada, o isomorfismo-dual e as fibras}\label{sec:observador}

\begin{definicao}[observador e a escada]\label{def:observador}
Um \textbf{observador} é um vector de invariantes
$I(x)$ lido sobre um corpo endereçado; cada observador define a
equivalência $x\sim_{I}y\iff I(x)=I(y)$. A \textbf{escada} é a
hierarquia estrita
\[
\boxed{
I_1=(E)\;\prec\;I_2=(E,\Phi)\;\prec\;I_3=(E,\Phi,\Phi_2)\;\prec\;\cdots
\;\prec\;I_\infty=\text{o espectro completo},
}
\]
com $\Phi_m=\sum_i (i{+}1)^m b_i$ (os momentos --- a fase $\Phi=\Phi_1$,
a curvatura $\Phi_2$) e o refinamento
$x\sim_{I_{k+1}}y\Rightarrow x\sim_{I_k}y$.
\[
\boxed{
\text{cada observador define uma resolução de identidade;}
\quad
\text{aumentar o observador refina as classes.}
}
\]
\end{definicao}

\begin{teorema}[a escada é estrita --- e a estrita é orgânica]\label{thm:escada}
Cada degrau da escada tem cegueira construtível, e ela vive nos dados
reais:
\begin{enumerate}
\item \textbf{$I_1$ não basta} (o Teorema da Absorção e a caçada):
existem corpos distintos com a mesma energia.
\item \textbf{$I_2$ não basta}: a \emph{dupla transposição espelhada}
--- trocar «ab»$\to$«ba» num sítio e «ba»$\to$«ab» noutro, à mesma
distância interna $d$ --- conserva o multiconjunto ($E$) e cancela a
fase: $\Delta\Phi=d(a{-}b)+d(b{-}a)=0$. \textbf{Todos os $4286$
registos do cristal admitem esta colisão em dados orgânicos} (o par
espelhado é ubíquo no texto), e $11$ de $20$ respostas do corpus banal
também.
\item \textbf{$\Phi_2$ resgata a família}:
$\Delta\Phi_2=d(b{-}a)\bigl((i'{+}j')-(i{+}j)\bigr)$ só zera com os
centros coincidentes --- separou as $4286$ e as $11$, todas.
\item \textbf{A divisão de trabalho, medida}: os $190$ pares
\emph{naturalmente} distintos do corpus banal separam-se todos já em
$I_1$ --- o degrau grosso resolve o mundo natural; o degrau fino
resolve o adversário.
\end{enumerate}
\[
\boxed{\text{conservar a norma não conserva a identidade.}}
\]
\emph{Prova (operacional, medida).} \code{tests/assinatura\_colisoes.js}
($4286/4286$; $\Phi_2$ separa $4286/4286$; $30\,000$ cadeias sem colisão
de $I_3$ --- orçamento dito); \code{tests/assinatura\_banal.js}
(a escada no orgânico: $190$ em $I_1$, $0$ em $I_2$/$I_3$, $0$ por
separar; $11$ espelhados só em $\Phi_2$). $\square$
\end{teorema}

\begin{teorema}[o teto do observador pelas oito leis]\label{thm:teto-oito}
Sejam $\Phi_0,\ldots,\Phi_7$ os \textbf{oito} momentos --- um por lei do
catálogo --- lidos no anel ($\mathbb{Z}_{65537}$). Então:
\begin{enumerate}
\item \textbf{Fecho em $8$}: se os oito momentos coincidem, não existe
diferença de suporte $\le8$ --- a matriz de Vandermonde das posições
tem $\det=\prod_{i<j}(x_i-x_j)$, e posições distintas menores que $p$
nunca o anulam no anel.
\item \textbf{E falha em $9$}: o padrão $\Delta^{8}$ (os binomiais
alternados $\binom{8}{k}(-1)^k$, suporte $9$) anula os oito momentos
--- escapa ao observador linear. A \textbf{energia} (quadrática)
apanha-o no caso genérico: a cruz outra vez --- o linear ordena, o
quadrático mede, e nenhum sozinho é o par.
\item \textbf{A torre continua}: acima do suporte $8$ sobe-se de andar
--- cada dobra dá momentos novos. O teto é do \emph{relógio} do
observador, não do objecto --- o mesmo gesto do relógio de Hurwitz
(Hurwitz~8 não é teto do objecto --- \emph{Corpo de Peano}; analogia de forma, não identidade).
\end{enumerate}
\emph{Prova (operacional, medida).} \code{tests/observador\_torre.js}:
Vandermonde sem zero em $200$ conjuntos de $8$ posições da torre;
$\Delta^{8}$ com os oito momentos \textbf{todos} iguais e
$\Delta E=12\,870\neq0$. $\square$
\end{teorema}

\begin{definicao}[isomorfismo-dual ordenado]\label{def:iso-dual}
Um mapa $f\colon X\to Y$ é \textbf{isomorfismo-dual ordenado} quando
preserva \emph{simultaneamente} o fecho do observador e a ordem:
\[
\boxed{
I(f(x))=I(x)
\qquad\land\qquad
x\prec_X y\iff f(x)\prec_Y f(y).
}
\]
A escada $(\Phi,\Phi_2,\ldots)$ é o \textbf{observador da ordem} ---
mede-a; não é a definição dela. A permutação admissível é a testemunha:
muda a posição física, preserva $I$ e a ordem própria (derivada do
endereço).
\end{definicao}

\begin{teorema}[as duas fusões --- e a fusão é multiplicação]\label{thm:fusao-mult}
A torre tem \textbf{duas} fusões, e cada uma conserva à sua maneira:
\begin{enumerate}
\item \textbf{Fusão aditiva} $\oplus$ (a dobra/clone --- Lei~4,
$T+T^{*}$): a energia \textbf{soma}, $E(u\oplus u)=2E(u)$, e a
retração $\pi$ devolve o original byte a byte.
\item \textbf{Fusão multiplicativa} $\otimes$ (corpos independentes):
a energia \textbf{multiplica},
\[
\boxed{E(u\otimes v)=E(u)\cdot E(v)}
\quad\text{exato em inteiros}
\]
--- a conservação multiplicativa da norma, $N(xy)=N(x)N(y)$: o
\textbf{cristal de Hurwitz}. $\Phi_0$ multiplica no anel. No viveiro a
mesma lei já estava escrita:
$\operatorname{lcm}\cdot\operatorname{gcd}=ab$.
\end{enumerate}
\[
\boxed{
\oplus\ \text{soma}\quad\longleftrightarrow\quad\otimes\ \text{multiplica}
\qquad\text{--- o par aditivo/multiplicativo da torre.}
}
\]
\emph{Prova (operacional, medida).} \code{tests/observador\_torre.js}
\S T3, \S T5: $50$ pares reais do cristal, igualdade exata; clone e
retração exatos. $\square$
\end{teorema}

\begin{teorema}[divisão $=$ fibra --- e a clássica emerge]\label{thm:divisao-fibra}
A \textbf{divisão} não se postula como aritmética: é a \textbf{fibra
inversa} da fusão,
\[
\boxed{
F^{-1}(z)=\{(x,y):F(x,y)=z\},
\qquad
I(z)=I(F(x,y))\ \text{(conservação)}.
}
\]
No caso multiplicativo, dado $z=u\otimes v$ e um dos fatores, a fibra
resolve-se componente a componente:
\[
\boxed{x_i=z_{ij}/v_j}
\quad\text{--- divisão \textbf{inteira} exata, consistente por todos os
$j$ (oito caminhos que concordam), devolvendo $u$ byte a byte;}
\]
e a energia divide: $E(u)=E(z)/E(v)$ exato.
\[
\boxed{\text{a divisão clássica emerge da fibra --- medida, não postulada.}}
\]
\emph{Prova (operacional, medida).} \code{tests/observador\_torre.js}
\S T4: $50$ casos reais, recuperação exata, oito caminhos concordantes,
energia a dividir sem resto. $\square$
\end{teorema}

\begin{observacao}[estatuto e quarentena]
Tudo aqui é operacional e medido; sem lei nova --- Lei~4 (a dobra),
Lei~7/Hurwitz (a conservação multiplicativa), o catálogo dos oito (o
teto). A \textbf{fusão de conceitos} saiu da quarentena \emph{pelos
dados}: o objeto apareceu no próprio cristal ($7$ pares
singular/plural e mais de uma centena de títulos duplicados --- casos
reais, não inventados). A fusão mede-se como \textbf{soma direta com
contorno}: $z$ guarda as duas partes intactas (a dualidade é a memória
da divisão --- fundir sem guardar os dois lados é dissipar), a
conservação fecha com o \textbf{termo de contorno exato}
\[
\boxed{E(z)=E(x)+E(y)+E_{\partial},\qquad E_{\partial}=E(\text{esqueleto})}
\]
--- dois caminhos que concordam ---, a fibra devolve as partes byte a
byte, e no corpus a fusão é \emph{transação declarada}
($R_{\mathrm{endereço}}=3$: dois saem, um entra) que \textbf{reverte a
$R=0$} --- todo passo reverte, ou paga. Medido:
\code{tests/fusao\_conceitos.js}. Fundir \emph{de facto} é curadoria
do dono.
O clone responde-se pela métrica $x\sim_{\mathcal O}y$: $I$ identifica
o corpo; o endereço conta a carga (o um e o dois); a admissão é do
Controle.
\end{observacao}

\section{Teorema da Absorção --- o estresse pelo endereço}\label{sec:absorcao}

O Metrónomo atesta a volta íntegra: $\lambda^{+}+\lambda^{-}=0$
(o Teorema do Metrónomo --- \emph{Corpo de Peano}). Falta a pergunta dual: quanto \textbf{custa}
uma indução? Se um passo induz $k$ elementos de um corpo de $n$
endereços ($\lambda^{+}=k$) e a volta lê o resíduo $R$ ($\lambda^{-}=R$),
o \textbf{estresse} é o excesso
\[
\boxed{e \;=\; R - k.}
\]
A lei exige $e=0$: a leitura não pode dissipar mais do que a indução
valeu. E a exigência decide-se pela \textbf{régua de leitura} --- que é o
tema deste teorema, não a robustez do objecto.

\begin{definicao}[duas leituras do mesmo corpo]\label{def:leitura-endereco}
Um corpo de $n$ elementos endereçados admite duas leituras da volta:
\begin{enumerate}
\item \textbf{por posição} --- comparam-se as listas \emph{ordenadas};
a posição de cada elemento na ordem derivada participa da medida;
\item \textbf{pelo endereço} --- cada elemento lê-se pelo seu endereço
próprio (a assinatura que \emph{cavalga o corpo}; \emph{Arquitetura},
teorema da assinatura), e a medida conta faltantes, excedentes e
alterados --- a ordem \textbf{não} participa, porque é derivada do
endereço.
\end{enumerate}
A segunda é a leitura \textbf{dual}: a involução endereço$\to$elemento
fecha sem passar pela lista.
\end{definicao}

\begin{teorema}[Absorção --- o estresse tem teto, e é zero]\label{thm:absorcao}
Na leitura pelo endereço, subindo a torre ($n$ por dobra --- Lei~4):
\begin{enumerate}
\item \textbf{Endereço preservado} (apagar, alterar conteúdo, alterar
categoria, reordenar): $e=0$ \textbf{exato} em todo andar ---
$\lambda^{+}+\lambda^{-}=0$ em contagens inteiras, sem um double.
\item \textbf{Endereço destruído} (o elemento vira ilegível): $e=1$
\textbf{constante} com variância zero --- o elemento destruído conta como
o \textbf{par} faltante$\oplus$excedente, a Lei~0 a nomear os dois nulos
($0=(+1)\oplus(-1)$): quem mata um endereço paga a morte \emph{e} o
nascimento do lixo, e mais nada.
\item \textbf{Teto dimensional}: a variância do estresse \textbf{não
cresce} pela torre em nenhuma classe de indução --- a medida escala
sem teto de dimensão a rasgá-la.
\end{enumerate}
Na leitura \emph{por posição}, a classe (2) \textbf{diverge}
$\sim n^{2}$ (média $n/2$): o elemento ilegível perde o lugar na ordem e
empurra tudo entre a sua posição e o fim. A dissipação é da
\textbf{régua}, não do objecto --- a ordem é derivada do endereço, e
\emph{a régua não transporta; a volta transporta}.
\[
\boxed{
\text{endereço}=\text{leitura dual: }e=0;\qquad
\text{posição}=\text{régua derivada: dissipa }\sim n^{2}.
}
\]
\emph{Prova (operacional, medida).} \code{tests/lyapunov\_torre.js}:
inteiro puro, gerador congruente determinístico, \textbf{uma} indução
com \textbf{duas} leituras (as réguas medem o mesmo objecto), $T=100$ por
célula, torre $134\to4286$ por dobra sobre o cristal recuperado. Medido:
$e=0$ exato em todas as células de endereço preservado; $e\equiv1$
($\sigma^{2}=0$) no endereço destruído lido pelo endereço; e por posição
$\sigma^{2}\cdot T^{2}$ sobe de $13\,435\,000$ para $16\,130\,133\,019$
quando $n$ sobe $32\times$ --- $n^{2}$ manda $1024\times$, mediu-se
$1200\times$, média $e\approx n/2$. Matriz de indução no medidor v2
(\code{tests/cristal\_volta.js} \S V3, as duas réguas na mesma cópia):
permutação e retoque da projeção \emph{absorvidos} ($R=0$); conteúdo,
categoria, remoção e duplicação $R=1$; endereço destruído --- posição
$R=1641$, endereço $R=2$, o par. $\square$
\end{teorema}

\begin{proposicao}[a involução dentro do ruído]\label{prop:involucao-ruido}
Num lote de $m$ induções independentes, duas que caem no
\textbf{mesmo endereço} aniquilam-se --- a alteração involutiva compõe
consigo própria e dá a identidade ($\mathrm{flip}\circ\mathrm{flip}=
\mathrm{id}$: \textbf{Lei~1}). O estresse aparece então \emph{negativo},
com probabilidade de colisão $\sim\binom{m}{2}/n$, que \textbf{cai} ao
subir a torre. Não é dissipação: é a involução a fechar dentro do próprio
ruído (medido no mesmo \code{lyapunov\_measure.js}: lote de $8$,
$e<0$ só por colisão, $e\to0$ com $n$).
\end{proposicao}

\begin{corolario}[variante admissível pelo endereço]\label{cor:variante-endereco}
No Controle de Histerese (o Controle de Histerese --- \emph{Corpo de Peano}), o
critério da vizinhança $\mathcal{V}$ ganha a forma medida:
\[
\boxed{
\text{variante que preserva o endereço}
\;\Rightarrow\;
\mathrm{RETAIN}\ \text{sem custo } (e=0);
}
\]
\[
\boxed{
\text{variante que destrói o endereço}
\;\Rightarrow\;
\mathrm{RETRACT}\ \text{com custo constante (o par), nunca crescente}.
}
\]
O sistema \textbf{distingue} variantes admissíveis das que violam o
invariante --- não vale «toda alteração é corrupção»: reordenar é
variante equivalente ($e=0$ por construção, porque a ordem é derivada), e
o reticulado continua inteiro.
\end{corolario}

\begin{observacao}[estatuto]
Teorema \textbf{operacional}: enuncia o que a medida fechou na classe
considerada (corpos endereçados; induções das cinco classes; torre
por dobra até $4286$). Não afirma robustez universal de todo corpo ---
afirma que \emph{o estresse é propriedade da régua de leitura}, e que a
leitura dual (endereço) o absorve. Sem lei nova: Lei~0 (o par dos nulos),
Lei~1 (involução no ruído), Lei~4 (a torre por dobra), Metrónomo (a
atestação $\lambda^{+}+\lambda^{-}=0$).
\end{observacao}

\section{Energia --- a conservação pelo Parseval dourado}\label{sec:energia}

\begin{definicao}[indução e meta-indução]\label{def:inducao}
O passo que transforma um corpo chama-se \textbf{indução} --- o mesmo
gesto da Lei~4: toda transformação é um passo de indução na torre. O
passo \emph{sobre} os passos --- a leitura dual que mede a indução e
valida o que ela conservou --- é a \textbf{meta-indução}: o Metrónomo a
ler a indução. A dualidade é a de sempre: a indução projecta
($\lambda^{+}$), a meta-indução lê ($\lambda^{-}$), e o que a
meta-indução \textbf{valida é a conservação de energia}.
\end{definicao}

\begin{definicao}[energia]\label{def:energia}
A \textbf{energia} de um corpo $x$ de componentes endereçadas é o
segundo momento
\[
\boxed{E(x)\;=\;\sum_i x_i^{2}}
\]
--- a \textbf{massa} da cruz (a massa da cruz: a segunda
coordenada é $\sum(x-c)^{2}$ --- \emph{Corpo de Peano}), pela régua ao quadrado ($E=m$ em p.u.;
\emph{Corpo estelar}). A energia é a \textbf{norma}: as leis de
conservação da norma do sistema são três nomes da mesma exigência ---
$N(xy)=N(x)N(y)$ (Hurwitz, o lado que conta), $|\det|=1$ (o fator de
potência) e $\lambda^{+}+\lambda^{-}=0$ (o Metrónomo) --- e
\textbf{Parseval é essa exigência lida na transformada}: a energia não
se perde ao atravessar, muda de domínio.
\end{definicao}

\begin{teorema}[Parseval multidimensão pelo teorema central]\label{thm:parseval-multi}
No anel da Lei~8 ($\mathbb{Z}_{65537}$, a transformada dourada
$F[x]_k=\sum_i x_i g^{ik}$ com $g$ de ordem $N$), para eixos de
dimensões $N_1,\ldots,N_d$ com raízes de ordens $N_i$:
\[
\boxed{
\sum_{\mathbf{k}} X_{\mathbf{k}}\,X_{-\mathbf{k}}
\;=\;
\Bigl(\prod_{i=1}^{d} N_i\Bigr)\cdot\sum_{\mathbf{i}} x_{\mathbf{i}}^{2},
}
\]
e o fator depende \textbf{só da dimensão total} $\prod N_i$, não da
fatoração em eixos. Derivação pelo \textbf{teorema central}
(o teorema central Gentil--Hurwitz, citado do \emph{Corpo estelar}):
\begin{enumerate}
\item \textbf{O par frente/volta.} Avaliar na raiz $g^{k}$ e na inversa
$g^{-k}$ é a ida e a reflexão --- $(X_k,X_{-k})$ é o par
$(\lambda^{+},\lambda^{-})$ na representação. O produto fechado
$\sum_k g^{(i-j)k}=N\,\delta_{ij}$ é \textbf{contar} (Hurwitz): a órbita
completa da raiz conta $N$ na diagonal e $0$ fora --- o corte discreto.
\item \textbf{O lado contínuo.} A mesma linha, integrada, é
$\int|f|^{2}=\int|F|^{2}$ com a medida de Lebesgue --- \textbf{integrar}
(Gentil). A ordem conserva-se porque a medida é $\sigma$-aditiva: o
degrau $\mathbb{Q}\to\mathbb{R}$ do teorema central transporta a
contagem para a integral sem perder a ordem.
\item \textbf{Multidimensão.} Os eixos compõem por \textbf{produto de
medidas} (Lebesgue/Fubini: a medida do produto é o produto iterado), e a
torre sobe por dobra (Lei~4) --- logo Parseval compõe eixo a eixo e o
fator multiplica: $\prod N_i$.
\end{enumerate}
\emph{Prova (operacional, medida).} \code{tests/cristal\_energia.js}
sobre o digest do cristal no anel ($N=2^{8}$, a acumulação do
\code{/AssinaturaOito}): a \textbf{mesma} dimensão $256$ fatorada como
$256$, $16\cdot16$ e $4^{4}$ --- a raiz de ordem $4$ é o período do $i$
(pental) --- dá o \textbf{mesmo} espectro de energia
($\Sigma X X_{-}=37\,222=256\cdot\Sigma A^{2}$ nos três caminhos); e a
raiz de ordem errada ($3^{128}$, ordem $512$) \textbf{quebra} a
identidade --- a asserção pode falhar. $\square$
\end{teorema}

\begin{corolario}[a verificação dá-se por energia]\label{cor:verifica-energia}
A volta de um corpo atesta-se \textbf{por energia}, sem comparação
analítica termo a termo:
\[
\boxed{
E(\text{fonte})=E(\text{reconstrução})
\quad\text{(inteiro exato)},
}
\qquad
\boxed{
\text{caminho espectral valida a conta (dois caminhos)}.
}
\]
A indução legítima \textbf{conserva} $E$; a que viola o invariante muda
$E$ e a meta-indução acusa. Medido
(\code{tests/cristal\_energia.js} \S E0, \S E5):

zero, agora lido na energia.
\end{corolario}

\section{A membrana de Dirac --- a raiz da transição}\label{sec:dirac}

A transformada universal (avaliação nas folhas $\sigma,\sigma^{\dagger}$;
realização inteira pela matriz companheira --- \emph{Corpo de Peano},
a ponte) tem agora a sua \textbf{construção de Dirac}: o operador que
fatora a transição do Metrónomo. A escolha do coordenador --- a raiz
quadrada, \emph{metade para cada lado} --- e a derivação que já existia:

\begin{teorema}[a borda é uma fatoração de Dirac]\label{thm:borda-dirac}
No próprio andar, $D=A_m$ satisfaz
\[
\boxed{D^{2}=m\,D+I}
\]
--- a equação da borda $x^{2}=mx+1$ \emph{é} a fatoração: o passo é a
raiz quadrada do salto composto $L=A_m^{2}$. E \emph{metade para cada
lado}: $A^{k}A^{k}=A^{2k}$ com $\det(A^{k})^{2}=\det(A^{2k})=+1$ ---
cada metade carrega $\det=(-1)^{k}$ e o par fecha
($\sigma\sigma^{\dagger}=-1$).
\emph{Medido:} \code{tests/dirac\_transicao.js} \S D1, $m=1..6$, exato.
\end{teorema}

\begin{teorema}[a obstrução, e o furo na torre]\label{thm:furo}
A raiz quadrada do \emph{próprio passo} não existe no andar:
\[
\boxed{
\nexists\,D\in M_2(\mathbb{Z})\ \text{com}\ D^{2}=A_m
}
\qquad
\text{(}\det(D)^{2}=-1\ \text{impossível; }\sigma^{\dagger}<0\text{)},
\]
--- a obstrução é o \textbf{preço do dual} $\sigma\sigma^{\dagger}=-1$.
E existe \textbf{inteira, um andar acima}: na dobra ($4\times4$),
\[
\boxed{
D=\begin{pmatrix}0&I\\A_m&0\end{pmatrix},
\qquad
D^{2}=A_m\oplus A_m,
\qquad
\det D=-1\ \text{(unidade: inversa inteira).}
}
\]
\emph{Medido:} busca exaustiva vazia no $2\times2$; $D^{2}=A\oplus A$
exato, $m=1..6$; $D^{-1}D=I_4$ (\S D2--D3).
\end{teorema}

\begin{teorema}[Dirac atua exatamente nas transições]\label{thm:dirac-transicao}
$D$ aterra na diagonal $\{(z,z)\}$ \textbf{se e só se} o par é uma
transição verdadeira do Metrónomo:
\[
\boxed{
D(x,y)\in\text{diagonal}
\iff
y=A_m x.
}
\]
As transições são o lugar próprio do operador --- e a diagonal é o seu
conjunto fixo de aterragem. O Inversor aceita $D$: a volta é inteira e
a escada $(E,\Phi,\Phi_2)$ regressa idêntica.
\emph{Medido:} $100/100$ transições na diagonal, $100/100$
não-transições fora; $30$ corpos com a escada restaurada (\S D4--D5).
\end{teorema}

\begin{teorema}[o salto espectral --- a dobra temporal]\label{thm:dobra-temporal}
$D$ satisfaz
\[
\boxed{D^{4}=m\,D^{2}+I}
\qquad\text{isto é}\qquad
\boxed{\text{mín}(D)=x^{4}-mx^{2}-1=\bigl(x^{2}\bigr)^{2}-m\bigl(x^{2}\bigr)-1:}
\]
o polinômio mínimo de $D$ é a \textbf{borda dobrada} pela substituição
$x\mapsto x^{2}$ --- a \textbf{dobra temporal}: dois meios-passos são um
tick. Os representantes espectrais de $D$ ($\pm\sqrt{\sigma}$,
$\pm\sqrt{\sigma^{\dagger}}$) vivem no corpo de grau $4$ --- \textbf{o
andar acima da torre}, acedido pela dobra. E a leitura adjunta fecha o
outro lado: $D^{\dagger}D=(mA+I)\oplus I$ --- o salto inteiro num lado,
a identidade no outro: \emph{metade para cada lado}, literalmente.
\emph{Medido:} \S D6, $m=1..6$, exato.
\end{teorema}

\begin{definicao}[membrana]\label{def:membrana}
Uma \textbf{membrana} é um corpo (qualquer função endereçada) que
atravessa a transformada. A sua \textbf{representação espectral no
corpo universal} obtém-se pelas \textbf{dobras temporais}: cada dobra
$x\mapsto x^{2}$ sobe um andar e a membrana ganha os representantes
desse andar (Teorema~\ref{thm:dobra-temporal}). Estatuto:
\emph{definição de realização} sobre os teoremas medidos --- a membrana
não é operador novo; é o corpo em trânsito pela fatoração.
\end{definicao}

\section{A morfologia --- o par $\delta\dashv\varepsilon$ e a torção}\label{sec:morfologia}

A morfologia do Universal não se importa de fora: \textbf{a construção
já estava no repo} (o mórfico do \emph{Catálogo}; \code{tests/toolkit.c}
\S K3; \code{tests/morfico\_ordem.c}; Porquês${}=\varepsilon$ /
5W2H${}=\delta$ no Conecthus), e o laboratório
(\code{tests/morfologia\_universal.js}, $14$:$0$) verificou-a pelo
critério decisivo:
\[
\boxed{
\text{morfologia válida}
\iff
\text{dual existe}\;\land\;\text{volta existe}\;\land\;\text{invariante conservado}.
}
\]

\begin{teorema}[o par morfológico]\label{thm:morf-par}
A dilatação e a erosão são o par \textbf{adjunto}
\[
\boxed{\delta(A)\subseteq C\iff A\subseteq\varepsilon(C)}
\qquad(4096/4096\ \text{pares, varredura completa}),
\]
e dele \textbf{emergem} --- não se postulam --- a abertura e o fecho
clássicos: $\alpha=\delta\circ\varepsilon\le\mathrm{id}\le
\varepsilon\circ\delta=\varphi$, ambos \emph{idempotentes}. A
\textbf{volta existe na fronteira admissível}:
$\delta(\varepsilon(A))=A$ \emph{exatamente} nos conjuntos
$B$-abertos ($29$ de $64$ no universo medido) --- $E(D(x))=x$ dentro
da fronteira, como a adjunção manda. O invariante é a \textbf{ordem do
parâmetro}: o raio \emph{soma}
($\operatorname{dil}_r\circ\operatorname{dil}_s=\operatorname{dil}_{r+s}$,
Minkowski), total e monótona no objeto. E o par \textbf{muda o corpo}:
$\delta$ cria ($A\subseteq\delta A$), $\varepsilon$ remove
($\varepsilon A\subseteq A$) --- a massa sobe e desce, medida.
\end{teorema}

\begin{teorema}[a torção]\label{thm:torcao}
A torção tem \textbf{duas realizações que concordam}, e nenhuma é
dilatação escondida:
\begin{enumerate}
\item \textbf{o esquilo} $J\colon(a,b)\mapsto(b,-a)$:
$J^{2}=-I$, $J^{4}=\mathrm{id}$, $|\det J|=1$, e a \textbf{norma
conserva exata} --- $J$ transporta a coordenada para o eixo dual sem
criar nem remover;
\item \textbf{a órbita do anel}: a torção $w$ ($3^{256}$ em
$\mathbb{Z}_{65537}$) fecha em $256$ e em nenhum divisor próprio ---
\emph{a base é a órbita da torção} (\emph{Catálogo}) --- e \textbf{o
dual do dual devolve o grupo}: ida pelos caracteres, volta pela órbita
inversa, resíduo $0$ --- o degrau \emph{discreto} de Pontryagin,
medido no anel.
\end{enumerate}
\[
\boxed{
\text{erosão/dilatação mudam o \textbf{corpo}};
\qquad
\text{a torção muda o \textbf{representante} sem o destruir.}
}
\]
E a decomposição da morfologia universal é
\[
\boxed{
\text{morfologia}=\underbrace{\delta/\varepsilon}_{\text{escala/representação}}
+\underbrace{J}_{\text{eixo/assinatura}},
}
\qquad
\text{corpo}\xrightarrow{\delta/\varepsilon}\text{representação}
\xrightarrow{J}\text{dual}\xrightarrow{\text{Inversor}}\text{corpo}.
\]
A torção participa do par de Clifford recém-fechado ($J^{2}=-I$ é o
rotor da Lei~2; $\mathrm{Cl}(1,1)$ dá a assinatura) --- a morfologia lê
a escala, a torção assina o eixo, o Inversor confere a volta, o
Metrónomo mede o custo.
\emph{Medido:} \code{tests/morfologia\_universal.js} \S M1--M6;
inventário re-atestado: \code{toolkit.c} ($8$ ok),
\code{morfico\_ordem.c} ($3$ ok), \code{universal.c} ($42$ ok),
\code{rede\_dual.js} ($28$:$0$).
\end{teorema}
