\begin{abstract}
\noindent Produto: a \textbf{partitura desenhada}. O pentagrama é o
\textbf{cone} (pera); a música é a \textbf{espiral} que sai da dinâmica
da vareta do Maestro e projecta a orquestra. Eixos: horizontal $=$
Metrónomo (tick, $\lambda^{+}+\lambda^{-}=0$); vertical $=$ altura no cone
--- não a diagonal. LilyPond grava; o tradutor desenha.
\end{abstract}

\section{Geometria (não a diagonal)}

\[
\boxed{
\mathrm{Conservat\acute{o}rio}
\;\xrightarrow{\mathrm{Pera}}\;
\text{vareta}
\;\rightarrow\;
P(t)
\;\rightarrow\;
\text{espiral}
\;\rightarrow\;
\text{orquestra}.
}
\]
\[
\boxed{
\text{pentagrama}=\text{pera}/\text{cone};
\quad
\text{música}=\text{espiral da vareta}.
}
\]
\begin{itemize}
\item \textbf{Horizontal} --- Teorema do Metrónomo: tick / leitura
métrica; involução $\lambda^{+}\leftrightarrow\lambda^{-}$.
\item \textbf{Vertical} --- cone (pera): altura no pentagrama; involução
$\nu$ (clave sol$\leftrightarrow$fá, figura$\leftrightarrow$silêncio).
\item \textbf{Maestro} --- Teorema $\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$:
projecta $\pi_k$; a vareta escreve a espiral; a orquestra realiza.
\end{itemize}

\section{Cadeia}

\[
\boxed{
\text{corpo}
\to\pi_k\to\Pi\to\text{Maestro}\to\text{vozes}\to\text{partitura}.
}
\]

\section{Condições (Peano / estelar)}

\begin{definicao}[Pentagrama $=$ pera / cone]
Cinco \verb|\rule| --- realização do \textbf{cone} (Def.~pera no Corpo de
Peano): base da batuta onde o conservatório entra no canal. Não é tabela
nem diagonal: é o suporte cónico da escrita.
\end{definicao}

\begin{definicao}[Música $=$ espiral]
A trajectória $P(t)$ na ponta da vareta --- periódica / espiralada ---
projecta os naipes. No tradutor a altura no cone realiza-se pela
\textbf{espiral de expoentes} ($_{\note}$, $\note$, $^{\note}$, \ldots)
--- a mesma subida $\mathrm{esp\_gira}$ do núcleo. $\Pi$ não inventa a
órbita: sai da dinâmica do Maestro sob o tick comum.
\end{definicao}

\begin{definicao}[Involução $\nu$ --- Lei~1 (eixo vertical)]
\[
\mathrm{figura}\;\overset{\nu}{\longleftrightarrow}\;\mathrm{silêncio},
\qquad\nu\circ\nu=\mathrm{id}
\]
(\verb|\N| $\leftrightarrow$ \verb|\Rr|). Clave sol
$\overset{\nu}{\longleftrightarrow}$ fá.
\end{definicao}

\begin{definicao}[Metrónomo --- Lei~6 (eixo horizontal)]
Tick comum; $\lambda^{+}+\lambda^{-}=0$ na volta fechada. Lê; não inventa
$\Delta$.
\end{definicao}

\begin{definicao}[Trial --- Lei~3]
Batuta canaliza $I\in\{-1,0,+1\}$ na vareta: grave / neutro / agudo.
\end{definicao}

\begin{definicao}[Maestro --- $\pi_k$]
Projecção sustentada no cone; ataque + silêncios ($\nu$) $=$ semibreve
no gabarito.
\end{definicao}

\begin{definicao}[Naipes / Lei~8]
Hurwitz $1,2,4,8$ e $\mathbb{Z}/8\mathbb{Z}$ --- a orquestra sob a
espiral.
\end{definicao}

\begin{definicao}[Página --- dimensão do relógio ($\pi_{\mathrm{página}}$)]
A página não interrompe o sistema: é um \textbf{andar de composição}
sob o mesmo Maestro.
\[
\boxed{
\mathrm{tick}
\rightarrow
\mathrm{compasso}
\rightarrow
\mathrm{partitura}
\rightarrow
\mathrm{página}
\rightarrow
\mathrm{torre}
}
\]
$\pi_{\mathrm{página}}$ projecta o bloco contínuo no suporte $P_n$; se
não cabe, $P_n\xrightarrow{\pi}P_{n+1}$ --- muda o suporte, não o
estado (relógio, espiral, voz, métrica, $\Pi$, linha dourada).
Partituras e tabelas quebram assim, sem pulo externo.
\end{definicao}
