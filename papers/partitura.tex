\begin{abstract}
\noindent Produto deste paper: a \textbf{partitura desenhada} do sistema ---
não uma tabela. Pentagrama por \verb|\rule|; figura por \verb|\note| (♪);
oito vozes no mesmo fluxo temporal. Cadeia:
\[
\boxed{
\text{corpo}
\to\pi_k\to\Pi\to\text{Maestro}\to\text{vozes}\to\text{partitura}.
}
\]
O pentagrama \textbf{não} é visualização de dados: é a realização musical
da assinatura. LilyPond (\code{partitura/emite.c}) só grava.
\end{abstract}

\section{Objectivo}

\[
\boxed{
\text{Tradutor}
\;\xrightarrow{\Pi}\;
\text{Partitura completa}
\;\xrightarrow{\text{tick}}\;
\{G_0,\ldots,G_n\}
\;\xrightarrow{\text{Maestro}}\;
\text{Orquestra}
}
\]
Maestro $=$ tempo comum; Batuta $=$ direcção; $\Pi=$ conteúdo; naipes $=$
vozes; pentagrama $=$ realização.

\section{Como o tradutor desenha}

\begin{definicao}[Pentagrama]
Cinco \verb|\rule{15.4cm}{0.32pt}| --- a régua do Reino. Metrónomo: uma
linha (gabarito LilyPond).
\end{definicao}

\begin{definicao}[Nota]
\verb|\note| no léxico $\mapsto$ U+266A na OTF $\mapsto$ Form XObject
(como \verb|\alpha| na Symbol).
\end{definicao}

\begin{observacao}
Pauta primeiro; \verb|\vspace| negativo; notas sobem ao meio do
pentagrama. Coluna comum $=$ barras verticais $|$ no mesmo sítio em
todas as vozes --- não uma tabela.
\end{observacao}
