\begin{abstract}
Um corpo possui operações e ordem. Não se postula $\mathbb{R}$ com
$+$ e $\times$: o corpo \textbf{constrói-se}. \textbf{A teoria vem no
fim:} antes, a torre de fundação
(\code{papers/torre\_fundacao.tex}) --- Corpos, Dinâmica, Topologia,
Hilbert/Clifford. A cadeia neste paper é
\[
\boxed{
\text{borda}
\;\to\;
\text{ordem}
\;\to\;
\text{massa}
\;\to\;
\text{contínuo}
\;\to\;
\text{torre}
\;\to\;
\text{retração }\pi_k
\;\to\;
\text{música}.
}
\]
O teorema central (\emph{Corpo estelar}): contar (Hurwitz) e integrar
(Gentil) são duais pela medida --- \emph{ponto de chegada}, não de
partida. O resultado deste paper não é só a
estrutura: a construção \textbf{admite realização dinâmica por retração
entre andares} e a sua \textbf{dual métrica para leitura}. O
\textbf{Teorema do Maestro}
(Teorema~\ref{thm:proj-maestro}) projecta ---
$\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$ executa
$\pi_k$ a cada tick. O \textbf{Teorema do Metrónomo}
(Teorema~\ref{thm:metronomo}) lê --- Lyapunov dualizado
$\lambda^{+}{+}\lambda^{-}=0$ nas interfaces das dobras, com a
direcção a entrar na base metálica e a transformada universal a
avaliar na transição. A música não é metáfora decorativa: é a
\textbf{linguagem de unificação} em que o modelo já sabia falar
(partitura$\to$maestro$\to$batuta$\to$instrumentos$\to$orquestra
traduz $\Pi\to\mathrm{Maestro}\to\pi_k\to G_i\to\mathcal{O}$).
O tradutor \texttt{.tex}$\leftrightarrow$PDF
(\emph{Arquitetura}) opera nesta torre: \code{MOVE} é a estrela; $\pi_k$ é a
retração. Milénios são leituras, não base
(\S\ref{sec:milenio-leitura}). O \textbf{Teorema da Histerese}
(\S\ref{sec:histerese}) deriva o conceito clássico
$(\theta_{\mathrm{high}},\theta_{\mathrm{low}})$ como memória da
borda sob a batuta --- sem Lei~$9$. O dual,
\textbf{Controle de Histerese} (\S\ref{sec:controle-histerese}),
selecciona variantes em $\mathcal{V}_k$ sem destruir o estado retido.
O \textbf{Teorema da Absorção} (\S\ref{sec:absorcao}) mede o
\emph{estresse} de uma perturbação pela torre: lido pelo
\textbf{endereço}, $e=R-k=0$ exato ($\lambda^{+}{+}\lambda^{-}=0$) e o
teto dimensional existe; a divergência $\sim n^{2}$ é da régua de
posição, não do objecto.
A \textbf{leitura complexa} (\S\ref{sec:pares-complexos})
formaliza o par $(R,I)$, deriva Euler (\S\ref{sec:euler-peano}),
e põe o \textbf{Metrónomo entre} as visões cartesiana e polar
(\S\ref{sec:metronomo-ponte}) --- mede o tick nas duas cartas; depois
a forma polar e a releitura de Lyapunov
(\S\ref{sec:lyap-dual-releitura}). Só então a \textbf{rede neural
dual} (\S\ref{sec:rede-dual}): Teoremas~\ref{thm:retencao-neural},
\ref{thm:rede-dual}; pesos $W=-I$ pela Lei~1; Hopfield $=$ memória da
volta, não inversão --- $\lambda^{+}+\lambda^{-}=0$ pela dualidade
reversível (Teorema~\ref{thm:metronomo}). Verificação:
\code{tests/corpo\_peano.c} §CP1--§CP40;
\code{tools/bench\_histerese.sh};
\code{tools/bench\_vizinhanca.sh};
\code{papers/dualsort.tex}.
\end{abstract}

\section{Família metálica e torre em $\mathbb{R}^{n}$}

\begin{definicao}[família metálica]\label{def:metal}
\[
A_m=\begin{pmatrix}m&1\\1&0\end{pmatrix},
\quad\det A_m=-1,
\quad
A_n=K[x]/(x^{2}-nx-1),
\quad
\sigma^{2}=n\sigma+1.
\]
Estaca $\sigma^{\dagger}=n-\sigma$; norma $N(\sigma)=-1$;
$\Delta=n^{2}+4$ (\code{tests/metalica.c}, \code{tests/fator.c}).
\end{definicao}

\begin{proposicao}[anti-autoadjuntos]\label{prop:anti}
Os $T$ com $T^{\dagger}=-T$ formam um corpo algébrico: em $2\times2$,
$\{aI+bJ\}\cong\mathbb{C}$; em $K[\sigma]$, a família metálica
($\sigma\sigma'=-1$). Hipótese: $-1$ não é quadrado em $K$
(\emph{Teoria}).
\end{proposicao}

\begin{teorema}[torre em $\mathbb{R}^{n}$]\label{thm:rn}
\[
A_{n+1}=A_n+A_n^{*},
\qquad
\dim A_n=2^{n}\dim A_0.
\]
Parte de $\mathbb{R}$ ($\dim A_0=1$). A ordem sobe por indução (total,
compatível com a soma); \textbf{não} é herdada da reta --- é produzida
pela dualidade. Em $\mathbb{R}^{n}$ a segunda coordenada da cruz é
$M_{ij}=\sum(x_i-c_i)(x_j-c_j)$.
\emph{Notação:} o índice $k$ da torre operacional
$d_k=2^{k+1}$ (com $d_0=2$) conta andares \emph{após} a primeira
dobra $A_0\to A_1$; não identificar $n$ de $A_n$ com $k$ de $d_k$ sem
essa convenção.
\end{teorema}

\section{A borda --- definição e unicidade de $bx^{a}$}

\begin{definicao}[borda]\label{def:borda}
A \textbf{borda} de índice $n\geq0$ é
\[
x^{2}=nx+1
\qquad\text{em }A_n=K[x]/(x^{2}-nx-1),
\]
equivalente a $x=n+1/x$ e a $x\,x^{\dagger}=-1$ com
$x^{\dagger}=-1/x=n-x$. Matriz companheira
$\bigl(\begin{smallmatrix}n&1\\1&0\end{smallmatrix}\bigr)$, $\det=-1$.
O corpo \textbf{materializa-se} na borda: o finito é $A_n$; a régua
(expansão) é infinita.
\end{definicao}

\begin{teorema}[a forma $bx^{a}$ é a única possível]\label{thm:forma-unica}
A equação da Lei~2
\[
-f \;=\; f^{-1}
\qquad\text{i.e.}\qquad
f\bigl(f(x)\bigr)=-x
\]
admite solução da forma $bx^{a}$, e essa forma é a \textbf{única} candidata
nas duas classes naturais do modelo:
\begin{enumerate}
\item \textbf{Analítica em $0$} com $f(0)=0$: o grau mínimo $m$
satisfaz $m^{2}=1$, logo $m=1$. O termo líder é $c_1 x$ com
$c_1^{2}=-1$ --- forma $bx^{a}$ com $a=1$ no líder.
\item \textbf{Euler-homogénea} em $\mathbb{R}_{>0}$ ($xf'=af$):
então $f(x)=b\,x^{a}$ exactamente.
\end{enumerate}
Na família $f(x)=bx^{a}$, a composição dá
$f(f(x))=b^{a+1}x^{a^{2}}$; exigir $-x$ força $a^{2}=1$.
\begin{itemize}
\item $a=+1$: $f(x)=bx$ e $b^{2}=-1$, logo
$\boxed{f(x)=\pm ix}$.
\item $a=-1$: $f(x)=b/x$ ($b\neq0$) e
$f(f(x))=x$ --- ramo \textbf{involutivo}, Lei~1, não a Lei~2.
É descartado para $f\circ f=-\mathrm{id}$ (não é ``$b=0$'').
\end{itemize}
Consequentemente
\[
\boxed{f\circ f=-\mathrm{id}
\;\Longrightarrow\;
f(x)=\pm ix}
\]
na classe considerada, sobre um corpo que contenha $i$.
Sobre $\mathbb{R}$ não há solução; sobre $\mathbb{C}$, as duas diferem
pelo sinal. Realização: $i^{2}=-1$, $i^{4}=1$ --- rotação de um quarto
(\code{tests/lei2.c}).
\end{teorema}

\begin{proof}
\textit{(Grau.)} $f(x)=\sum_{k\geq m}c_k x^{k}$ dá
$f(f(x))=c_m^{m+1}x^{m^{2}}+\cdots$. Igualar a $-x$ força $m^{2}=1$,
donde $m=1$ e $c_1^{2}=-1$.

\textit{(Homogeneidade.)} $xf'=af$ $\Rightarrow$ $f=bx^{a}$.

\textit{(Família.)} $f(f(x))=b^{a+1}x^{a^{2}}$. Se $a^{2}=1$ e $a=+1$:
$b^{2}=-1$. Se $a=-1$: $f(x)=b/x$ e $f(f(x))=x$ para todo $b\neq0$
--- nunca $-x$. Controlo: só expoentes com $p^{2}=q^{2}$ passam o
teste de $a=1/a$; $f(f(x))=-x$ exacto em $\mathbb{Z}[i]$ para
$\pm i$.
\end{proof}

\begin{proposicao}[o ramo $a=-1$ é a passagem pelo ponto fixo]\label{prop:a-1}
$f(x)=b/x$ realiza a estaca $\nu(x)=-1/x$ (a menos de escala):
$\nu\circ\nu=\mathrm{id}$, período $2$ --- \textbf{Lei~1}. Percorre a
órbita $\{0,\infty\}$: dividir \emph{por} zero é o $\infty$; a estaca
troca $0\leftrightarrow\infty$ e do outro lado \textbf{continua}.
Não é degeneração: é a transição dimensional no infinito
(\emph{Corpo estelar}, divisão do zero).
\[
\boxed{
a=-1:\ \text{Lei~1 (involução)};
\qquad
a=+1:\ \text{Lei~2 (rotação }\pm i\text{)}.
}
\]
A leitura topológica da Lei~1 (Poincaré) e a do centro da Lei~2
(Yang--Mills, $P$ vs $NP$) são derivações --- \S\ref{sec:milenio-leitura}
--- não axiomas deste corpo.
\end{proposicao}

\begin{proposicao}[costura no centro: as leis, não o milénio]\label{prop:centro}
A passagem pelo ponto fixo $\{0,\infty\}$ abre os \textbf{dois nulos}
operacionais (Lei~0: $0=(+1)\oplus(-1)$, $0^{\dagger}=\infty$). No
centro $0$ --- ponto fixo do bidual $K^{**}=K$ --- a Lei~2
realiza-se como rotor / parada. Dividir \emph{por} zero (o $\infty$)
é a Lei~1 (ramo $a=-1$). Dividir \emph{o} zero é a Lei~0 a nomear o
par, com a Lei~2 a fixar o centro. A transição dimensional no infinito
\textbf{é} essa costura:
\[
\boxed{
\text{Lei~1 atravessa};\quad
\text{Lei~2 fixa o centro};\quad
\text{Lei~0 nomeia os nulos}.
}
\]
\end{proposicao}

\begin{teorema}[unicidade sob $f^{(n)}=f^{-1}$]\label{thm:unica-borda}
Fixe $n\geq1$. Toda solução Euler-homogénea de
\[
f^{(n)}=f^{-1}
\]
é da forma $f(x)=b\,x^{a}$ (Teorema~\ref{thm:forma-unica}), e o
expoente $a>0$ é \textbf{único}:
\[
a-n=\tfrac1a
\;\Longleftrightarrow\;
a^{2}-na-1=0
\;\Longleftrightarrow\;
a=\sigma_n=\tfrac{n+\sqrt{n^{2}+4}}{2}.
\]
Em particular $a=\sigma_n$ \textbf{é a borda}. A outra raiz
$\sigma_n^{\dagger}=-1/\sigma_n<0$ descarta-se (polo em $0$;
\code{tests/aurea.c}, \code{tests/pisotbase.c}).
\end{teorema}

\begin{proof}
Homogeneidade $\Rightarrow$ $f=bx^{a}$. A $n$-ésima derivada tem
expoente $a-n$; a inversa tem expoente $1/a$. Igualar:
$a(a-n)=1$, isto é $a^{2}-na-1=0$. Discriminante $n^{2}+4>0$:
duas raízes. Só $\sigma_n>1$ dá $a>0$ com $f$ crescente e
inversível em $(0,\infty)$.
\end{proof}

\begin{corolario}[Pisot]\label{thm:pisot}
Toda razão metálica é unidade quadrática de Pisot:
$|\sigma_n|>1$, $|\sigma_n^{\dagger}|<1$, logo
$\|\sigma^{k}\|\to0$ como $|\sigma'|^{k}$ (vazamento zero;
\code{tests/pisotbase.c}).
\end{corolario}

\begin{definicao}[zeta dinâmica]\label{def:zeta}
\[
\zeta(x)=\frac{1}{\det(I-xC)}=\frac{1}{1-mx-x^{2}},
\quad
C=\begin{pmatrix}m&1\\1&0\end{pmatrix}.
\]
$\zeta(0)=1$: unidade de $\mathbb{Z}[[x]]$. Coeficientes
$c_n=mc_{n-1}+c_{n-2}$ (Fibonacci se $m=1$); traços
$t_k=c_k+c_{k-2}$ (Lucas se $m=1$). Fecho:
$\zeta=1/\nu(\beta)$ --- polos $=\sigma$
(\code{tests/zetadin.c}).
\end{definicao}

\section{Massa ordenada e max-cut --- o corpo entre dimensões}

\begin{proposicao}[massa na borda]\label{prop:massa}
A segunda coordenada da cruz é o segundo momento:
$nc^{2}-\sum xx^{\dagger}=\sum(x-c)^{2}$. Em $\mathbb{R}^{n}$ é a
matriz $M_{ij}$; a massa escalar é o seu traço
(\emph{Corpo estelar}). A massa é \textbf{ordenada}: no bidual a ordem
existe, e uma ordem sobre $n$ partes precisa de $n$ componentes --- o
escalar colapsa-a.
\end{proposicao}

\begin{teorema}[max-cut $=$ massa]\label{thm:maxcut}
Ordenar e cortar são duais (\emph{Dualsort}). Com elementos em
$+1$ e $-1$, $|A|+|B|=n$:
\[
\mathrm{corte}=|A|\,|B|,
\qquad
\mathrm{massa}=n-\frac{(|A|-|B|)^{2}}{n},
\]
máximos no mesmo argumento. O corte é a \textbf{sequência de lados}
--- corpo ordenado materializado na borda, uma componente por vértice
do reticulado --- não o escalar $|A||B|$. Meia volta
$i\mapsto(i+n/2)\bmod n$ constrói a bipartição
(\code{tests/maxcut.c}).
\[
\boxed{
\text{borda}=\text{suporte};\quad
\text{massa}=\text{corte ordenado};\quad
\text{corpo entre dimensões}=\text{ordem na borda}.
}
\]
\end{teorema}

\section{Contínuo, Cantor e Peano}

\begin{proposicao}[Gentil--Hurwitz via Lebesgue]\label{thm:central-peano}
Conforme o \textbf{teorema central} do \emph{Corpo estelar}
(Teorema~central ali; citado, não reprovado aqui): contar (Hurwitz) e
integrar (Gentil) são duais pela soma reversível
\[
\int_a^{b}f+\int_{f(a)}^{f(b)}f^{-1}=b\,f(b)-a\,f(a).
\]
O salto $\mathbb{Q}\to\mathbb{R}$ é este cruzamento. Ordem e contínuo
recuperam-se aqui; não se postulam.
\[
\boxed{
\text{Hurwitz (contar, discreto)}
\;\longleftrightarrow\;
\text{Gentil (integrar, contínuo)}.
}
\]
A bijeção dual é a \textbf{estrela} $\nu$ com $\nu\circ\nu=\mathrm{id}$
(involução, Lei~1 no nível da medida): liga sem fundir, resíduo~$0$.
\textbf{Destino:} Teorema Central $\Rightarrow$ retração $\pi_k$
$\Rightarrow$ Teorema do Maestro (\S\ref{sec:maestro-thm})
$\Rightarrow$ controlo / rede dual (\S\ref{sec:rede-dual})
$\Rightarrow$ realização musical. Estatuto aqui: \emph{citação / eixo}.
\end{proposicao}

\begin{proposicao}[leitura operacional do Teorema Central na torre]\label{prop:central-operacional}
O Teorema Central (Prop.~\ref{thm:central-peano}) dualiza
contar$\leftrightarrow$integrar. Na torre operacional, a mesma
dualidade realiza-se em \textbf{quatro} camadas sem as colar:
\begin{enumerate}
\item \textbf{Estrutura.} Retração $\pi_k$ (Def.~\ref{def:pi-k}): desce
o andar --- o ``contar'' do nível superior projecta-se no inferior.
\item \textbf{Dinâmica / métrica.} Maestro projecta
(Teorema~\ref{thm:proj-maestro}) via
$\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$;
Metrónomo atesta $\lambda^{+}+\lambda^{-}=0$
(Teorema~\ref{thm:metronomo}) --- o ``integrar'' da volta,
resíduo~$0$. A partitura $\Pi$ é o \emph{quê}
(Def.~\ref{def:partitura}): transporte, não fonte das leis.
\item \textbf{Controlo.} Histerese e rede dual
(Teoremas~\ref{thm:controle-histerese},~\ref{thm:rede-dual};
Prop.~\ref{prop:fecho-conj-exp}): $F_H=D\circ F_P^{-1}\circ D^{-1}$
é a \emph{mesma forma} da estrela (involução $D$, soma de expoentes
nula), realizada como controlador --- não como substituto do
Teorema Central.
\item \textbf{Música.} Conservatório$\to$Pera$\to$Batuta
(\S\ref{sec:musica}): o canal geométrico que transporta $\pi_k$ e o
comando $I\in\{-1,0,+1\}$ (batuta) --- sombra do conservatório, não
fonte do teorema.
\end{enumerate}
\[
\boxed{
\nu_{\mathrm{Central}}
\;\sim\;
\bigl(\pi_k,\;\lambda^{+}+\lambda^{-}=0,\;F_H,\;
\mathrm{Conservat\acute{o}rio}\!\to\!\mathrm{Pera}\!\to\!\mathrm{Batuta}\bigr).
}
\]
\emph{Proibições:} não identificar rede neural com Gentil--Hurwitz;
não identificar batuta com $I$ (a batuta \emph{canaliza} $I$);
a música \emph{realiza} o canal sob a atestação que o Central exige.
\end{proposicao}

\begin{proposicao}[Hurwitz~8 não é teto do objecto]\label{prop:sem-teto}
O grau oito do lado discreto (norma bilinear) \textbf{fecha em~$8$ e
falha em~$16$}: isso classifica o relógio da contagem (Hurwitz), não
põe parede no objecto. O lado contínuo (Gentil / Lebesgue) \textbf{não
herda} esse período --- a torre operacional $d_k=2^{k+1}$ sobe sem
esse tecto (\code{tests/estelar\_completo.c} §C3; \code{tests/traduz\_pi\_dim.js}:
dim~$128$ medida). Distinguir três coisas:
\begin{center}\small
\begin{tabular}{@{}lll@{}}
nome & o que limita & o que \textbf{não} limita \\
\hline
Hurwitz~$8$ & período do relógio discreto & o objecto / a torre \\
$\operatorname{Ind}^{8}=\mathrm{id}_{\mathcal L}$ & catálogo das leis & a dimensão $d_k$ \\
$\mathrm{TORRE\_NTT\_MAX}=4096$ & selo NTT Gentil ($Z_{65537}$) & a torre operacional \\
\end{tabular}
\end{center}
\[
\boxed{
\text{Hurwitz~8}=\text{período do relógio};\quad
\text{Gentil continua};\quad
\text{NTT}=\text{cap do selo, não da torre}.
}
\]
\end{proposicao}

\begin{definicao}[corpo de Cantor]\label{def:cantor}
Lado discreto/aditivo do corpo de Alonzo ($f_c(z)=z^{2}+c$): o shift
$\theta\mapsto2\theta$, dual a $z\mapsto z^{2}$ (Julia). Os golden
\[
\mathrm{cantor}=\varphi,\qquad
\mathrm{julia}=\psi,\qquad
\varphi\psi=-1
\]
são o par metálico $m=1$ (\code{tests/cantor\_julia\_dual.c}).
\end{definicao}

\begin{definicao}[endereçamento de Peano]\label{def:peano}
$\Lambda_K=(\mathbb{Z}/3^{K}\mathbb{Z})^{2}$, $|\Lambda_K|=9^{K}$,
trial $\{-1,0,+1\}$. Encher$\leftrightarrow$contar $=$
Gentil$\leftrightarrow$Hurwitz no endereço. A curva clássica é o
limite $K\to\infty$; $\phi_K$ finito não se identifica com ela
(\code{tests/peano\_dual.c}).
\end{definicao}

\begin{definicao}[dual de Peano]\label{def:nu}
$\nu(\phi_K)=\phi_K^{-1}$, $\nu(\phi_K^{-1})=\phi_K$, logo
$\nu^{2}=\mathrm{id}$. Distinto de $\phi_K^{-1}\circ\phi_K=\mathrm{id}$
(reversibilidade da bijeção).
\end{definicao}

\section{Soma e produto na torre de dimensões}

\begin{definicao}[operações entre andares]\label{def:somaprod}
\begin{enumerate}
\item \textbf{Soma} (clone / tecidos):
$T_{k+1}=T_k+T_k^{*}$, $d_{k+1}=2d_k$, $d_0=2$, logo $d_k=2^{k+1}$.
\item \textbf{Produto} (reprodução / viveiro):
$a\otimes b:=\operatorname{lcm}(a,b)$,
$a\otimes^{*}b:=\operatorname{gcd}(a,b)$,
\[
\boxed{\operatorname{lcm}(a,b)\,\operatorname{gcd}(a,b)=ab.}
\]
$\mathbb{R}^{a}\vee\mathbb{R}^{b}=\mathbb{R}^{\operatorname{lcm}(a,b)}$
é rótulo do nível, não extensão de corpos.
\end{enumerate}
\end{definicao}

\begin{teorema}[reticulado divisorial]\label{thm:reticulo}
$a\le b$ sse $a\mid b$ em $\mathbb{N}_{>0}$: ordem parcial
independente da dualidade. $\vee=\operatorname{lcm}$,
$\wedge=\operatorname{gcd}$; distributivo (§CP12). Em
$D_N=\{d:d\mid N\}$, $\nu_N(d)=N/d$ é antiautomorfismo:
inverte a ordem, troca $\vee$ com $\wedge$.
\end{teorema}

\begin{observacao}
A dualidade \textbf{não cria} a ordem; inverte uma ordem já existente
em $D_N$. Reticulado e dualidade são estruturas relacionadas, não a
mesma afirmação.
\end{observacao}

\begin{proposicao}[interface cristalina]\label{prop:iface}
$\mathcal{O}=\{1,2,3,4,6\}$.
$\operatorname{lcm}(2,3)=6$,
$\operatorname{lcm}(2,3,4)=12$,
$\operatorname{lcm}(12,8)=24$.
A família de interfaces é
$\iota_k=6\cdot2^{\lfloor k/3\rfloor}$ --- valores da
\emph{interface}, não índices de leis.
\end{proposicao}

\section{Retração e o Teorema do Maestro}\label{sec:maestro-thm}

A torre sobe por $T+T^{*}$ (Def.~\ref{def:somaprod}). O dual que
\textbf{desce} é a retração $\pi_k$. O Teorema Central
(Prop.~\ref{thm:central-peano}) dá a passagem estrutural
Gentil\,$\leftrightarrow$\,Hurwitz; $\pi_k$ é essa passagem entre
andares; o maestro é a mesma passagem \textbf{realizada no tempo}.
\[
\boxed{
\text{Gentil--Hurwitz}
\;=\;
\text{passagem estrutural};
\qquad
\text{Maestro}
\;=\;
\text{passagem realizada no tempo}.
}
\]
\[
\boxed{
\text{Metrónomo}
\;=\;
\text{passagem lida (métrica)}.
}
\]
\[
\boxed{
\text{Teorema Central}
\;\longrightarrow\;
\pi_k
\;\longrightarrow\;
\bigl(\text{Maestro},\,\text{Metrónomo}\bigr)
\;\longrightarrow\;
\text{Música}.
}
\]

\begin{definicao}[Projecção dimensional $\pi_k$]\label{def:pi-k}
Em cada andar $k$ da torre operacional
(Def.~\ref{def:somaprod}: $T_{k+1}=T_k+T_k^{*}$,
$d_{k+1}=2d_k$), a \textbf{projecção dimensional} é o esquecimento
dual da indução --- uma fase / um factor por degrau:
\[
\boxed{
\pi_k\colon A_{k+1}\longrightarrow A_k,
\qquad
\pi_k(x\oplus y^{*})=x.
}
\]
O embebimento canónico $\iota_k\colon A_k\hookrightarrow A_{k+1}$,
$\iota_k(x)=x\oplus 0^{*}$, faz de $\pi_k$ uma \textbf{retração}:
\[
\boxed{
\pi_k\circ\iota_k=\mathrm{id}_{A_k}.
}
\]
Em dimensão: $d_{k+1}/2=d_k$. No catálogo de fases, a mesma forma
desce uma $-$alidade por passo (\code{tests/milenio\_trial.c} §M8).
No cone: sombra no hiperplano (\code{tests/sombra\_cone.c}). Estatuto:
\emph{definição} --- já na torre.
\end{definicao}

\begin{teorema}[Maestro --- realização dinâmica do Teorema Central]\label{thm:proj-maestro}
Seja $\pi_k\colon A_{k+1}\to A_k$ a retração entre andares consecutivos
(Def.~\ref{def:pi-k}), com $\pi_k\circ\iota_k=\mathrm{id}_{A_k}$.
Então uma realização dinâmica dessa retração implementa-se por um
maestro constituído por relógio (tick), canal linear de batuta e
partitura $\Pi$:
\[
\boxed{
\mathcal{P}_k
\;=\;
\mathrm{tick}
\;\circ\;
\mathrm{batuta}
\;\circ\;
\Pi,
}
\]
realizando $\pi_k$ a cada tick. O maestro \textbf{não} cria a
estrutura de $A_k$: realiza a retração da estrutura superior para a
inferior.
\[
\boxed{
\mathrm{maestro}
\;=\;
\text{realização dinâmica de }\pi_k.
}
\]
\[
\boxed{
\text{O maestro não compõe o corpo: projecta-o.}
}
\]
Três afirmações distintas:
\begin{enumerate}
\item \textbf{Retração} --- $\pi_k\circ\iota_k=\mathrm{id}$,
$d_{k+1}/2=d_k$ (álgebra da torre).
\item \textbf{Conservação} --- se a classe Gentil/cruz admite,
$|X_{k+1}|_{\mathrm{Gentil}}=|X_k|_{\mathrm{musical}}$ (mesmo
invariante do Teorema Central).
\item \textbf{Realização musical} --- batuta e $\Pi$ transportam
$\pi_k$; não são operador novo.
\end{enumerate}
\[
\boxed{
\text{retração}
\;\neq\;
\text{conservação}
\;\neq\;
\text{realização musical}.
}
\]
Correspondência:
\begin{center}\small
\begin{tabular}{@{}lll@{}}
Teorema Central & na torre & no tempo / interface \\
\hline
Gentil (integrar) & andar $A_{k+1}$ & conservatório \\
Hurwitz (contar) & andar $A_k$ & música / orquestra \\
bijeção / estrela & $\pi_k$ & maestro $(\mathrm{tick},\mathrm{batuta},\Pi)$ \\
atestação $\lambda^{+}+\lambda^{-}=0$ & Metrónomo & resíduo da volta \\
controlo $(P,\mathcal{H},H)$ & rede dual & escolha $F_P$/$F_H$ via $I$ \\
\end{tabular}
\end{center}
\emph{Prova.} (1)~Def.~\ref{def:pi-k}. (2)~Prop.~\ref{thm:central-peano}
e cruz dos tecidos, na classe. (3)~$\mathcal{P}_k$ não acrescenta
operador: só compõe interfaces que executam $\pi_k$. A geometria
Conservatório$\to$Pera$\to$Batuta (\S\ref{sec:musica}) é a realização
concreta do canal, não a fonte do teorema. Controlo:
Prop.~\ref{prop:central-operacional}, \S\ref{sec:rede-dual}.
\end{teorema}

\begin{teorema}[Metrónomo --- realização métrica da dinâmica]\label{thm:metronomo}
O Teorema do Maestro (Teorema~\ref{thm:proj-maestro}) realiza $\pi_k$
no tempo: \textbf{projecta}. O dual dessa dinâmica --- a
\textbf{leitura mensurável do relógio} --- é o Teorema do Metrónomo.
Não compete com o Maestro: é a realização métrica do \emph{mesmo} eixo.
\[
\boxed{
\begin{array}{ccc}
\text{Teorema Central} & & \\
\downarrow & & \\
\text{retração }\pi_k & & \text{invariante} \\
\downarrow & & \downarrow \\
\text{Maestro} & & \text{Metrónomo} \\
\text{realização dinâmica} & & \text{realização métrica} \\
\downarrow & & \downarrow \\
\text{música} & & \text{Lyapunov dualizado}
\end{array}
}
\]
O relógio discreto da torre tem leitura dual: expansão e reflexão.
Se um passo emite $\lambda^{+}$, a volta dual dá $\lambda^{-}=-\lambda^{+}$.
Na classe operacional considerada, numa volta fechada:
\[
\boxed{\lambda^{+} + \lambda^{-} = 0},
\qquad
\boxed{\text{volta fechada}\;\Longleftrightarrow\;\lambda_{\mathrm{med}}=0}.
\]
O metrónomo realiza essa estrutura no tempo: cada tick marca a frente;
a reflexão mede a volta. O Lyapunov dualizado \textbf{atesta} a
reversibilidade do passo (\emph{Medida}, Teorema operacional;
\code{tests/lyapunov\_refletido.c}). Não é o Lyapunov clássico
(só $\lambda^{+}$): é a leitura métrica dualizada do passo.
\[
\boxed{
\text{O maestro projecta;}
\;
\text{a batuta canaliza;}
\;
\text{o metrónomo lê;}
\;
\text{o Lyapunov dualizado atesta.}
}
\]
\begin{enumerate}
\item \textbf{Base metálica --- direcção discreta.} A direcção
$I\in\{-1,0,+1\}$ entra na família metálica
(Def.~\ref{def:metal}, Def.~\ref{def:borda}):
\[
A_m
\;\longrightarrow\;
\sigma^{2}=m\sigma+1
\;\longrightarrow\;
\text{passo/expansão}
\;\longrightarrow\;
\text{reflexão dual}
\;\longrightarrow\;
\pm\lambda
\;\longrightarrow\;
\text{tick}.
\]
Nas interfaces das dobras ($\Delta=m^{2}+4$), $A_m$ é a régua discreta
que o relógio mede; $\sigma\sigma^{\dagger}=-1$ apresenta frente$+$dual
sem identificar $\lambda$ com instabilidade clássica. A batuta só
canaliza $I$ (Def.~\ref{def:batuta}).
\item \textbf{Transformada universal --- ponte de representação}
(citação, não fundamento novo):
\[
\text{base metálica}
\;\xrightarrow{\;\mathcal{T}\;}\;
\text{representação do relógio}
\;\xrightarrow{\;\mathcal{T}^{-1}\;}\;
\text{leitura dual},
\]
com $\mathcal{T}$ a avaliação nas raízes da borda já definida
(\emph{Corpo estelar}; \code{tests/universal.c}): $F[f]=f(\sigma)$
na transição. Não é DFT nem $\sqrt N$ no objecto
($\sigma\sigma'=-1$).
\end{enumerate}
Hierarquia:
\begin{center}\small
\begin{tabular}{@{}lll@{}}
camada & objecto & função \\
\hline
fundamento & Teorema Central & contar $\leftrightarrow$ integrar \\
estrutura & $\pi_k$ & retração \\
dinâmica & Maestro & realiza a projecção \\
tempo & Metrónomo & realiza o tick / lê \\
espaço & Morfologia $(W,H)$ duais & Lei~4 / $\operatorname{Ind}^{8}$+Estrela \\
página & $\pi_{\mathrm{página}}$ & andar do relógio \\
métrica & Lyapunov dualizado & atesta a volta \\
base discreta & $(A_m,\sigma_m)$ & régua metálica \\
representação & $\mathcal{T}$ & transforma a leitura \\
realização & música / partitura gráfica & resultado observado \\
\end{tabular}
\end{center}
\[
\boxed{
\underbrace{\text{Maestro}}_{\text{projecta}}
\;\leftrightarrow\;
\underbrace{\text{Metrónomo}}_{\text{lê}};
\qquad
\text{Lyapunov dualizado}=\text{atestação métrica da volta}.
}
\]
\emph{Prova.} (1)~Teorema operacional (\emph{Medida}): espectro
$\pm\lambda$ na classe reversível; fiabilidade $=$ volta com
resíduo~$0$ (critério operacional, não lei de todo sistema dinâmico).
(2)~Def.~\ref{def:metal}--\ref{def:borda}: direcção na interface
$\Delta$, não inventada pelo metrónomo. (3)~$\mathcal{T}$ citada:
instrumento de leitura. Separação:
projectar $\neq$ canalizar $\neq$ ler $\neq$ atestar.
\end{teorema}

\begin{teorema}[Morfológico --- coordenadas duais do suporte]\label{thm:morfologico}
O par Maestro\,/\,Metrónomo realiza no \emph{tempo} a dualidade
projecção\,/\,leitura. O \textbf{corpo morfíco} realiza a mesma
dualidade no \emph{espaço}, com \textbf{duas coordenadas duais}
$(W,H)$ --- sem Lei~$9$:
\[
\boxed{
(W,H)
\;=\;
\text{coordenadas duais do suporte},
\qquad
\mathcal{M}(\Pi)\sim\Pi.
}
\]
\begin{enumerate}
\item \textbf{Largura $W$ --- Lei~4 (dobra).}
$T+T^{*}$, $d\mapsto 2d$, interfaces $\Delta=m^{2}+4$:
\[
\boxed{
\mathcal{M}_{W}=\mathcal{D}_{W}\circ\mathcal{C}_{W}^{-1}
\;:\;
\text{preenche }[0,W]
}
\]
(eixo do Metrónomo; pauta e ticks na régua COL).
\item \textbf{Altura $H$ --- $\operatorname{Ind}^{8}$ / Estrela.}
O fecho do catálogo $\operatorname{Ind}^{8}=\mathrm{id}_{\mathcal L}$
\textbf{valida} a coordenada vertical: a pilha de vozes é fusão por
estrela (Def.~\ref{def:orquestra}),
$\mathcal{O}=\mathrm{Estrela}(\{G_i\},\,\mathrm{Maestro})$.
Sem interface estrela a altura \textbf{não} dilata --- não há suporte
vertical comum.
\[
\boxed{
\mathcal{M}_{H}=\mathcal{D}_{H}\circ\mathcal{C}_{H}^{-1}
\;:\;
\text{preenche }[0,H]
\quad\text{(requer Estrela)}.
}
\]
\end{enumerate}
Ordem da torre no suporte:
\[
\boxed{
W\ (\mathrm{Lei~4})
\;\longrightarrow\;
H\ (\operatorname{Ind}^{8}/\mathrm{Estrela}).
}
\]
\[
\boxed{
(\mathrm{Maestro},\,\mathrm{Metr\acute{o}nomo})
\;\longleftrightarrow\;
(\mathcal{D},\,\mathcal{C})
\;\longleftrightarrow\;
(W,\,H).
}
\]
\emph{Prova.} (1)~Lei~4 já passa entre escalas no plano horizontal.
(2)~$\operatorname{Ind}^{8}$ fecha o anel; a orquestra só empilha sob
Estrela --- a coordenada $H$ herda essa validação. (3)~$(W,H)$ são
duais como $(\lambda^{+},\lambda^{-})$: dilatar numa e contrair na outra
reencontra a mesma realização. Catálogo intacto: leis $0$--$7$;
$\operatorname{Ind}^{8}$ é fecho, não nona lei.
\end{teorema}

\begin{center}\small
\begin{tabular}{@{}llll@{}}
eixo & lei / fecho & interface & objecto \\
\hline
tempo & Maestro / Metrónomo & relógio & projecta / lê \\
$W$ (largura) & Lei~4 (dobra) & $\Delta$ & dilata / contrai horizontal \\
$H$ (altura) & $\operatorname{Ind}^{8}$ & Estrela & dilata / contrai vertical \\
volta & $\lambda^{+}+\lambda^{-}=0$ & --- & $\mathcal{M}(\Pi)\sim\Pi$ \\
\end{tabular}
\end{center}

\section{O corpo operacional $\mathcal{C}_K$}

\begin{definicao}[corpo operacional de Peano]\label{def:corpo}
\[
\mathcal{C}_K
:=
\bigl(\mathbb{N}_{>0},\otimes,\otimes^{*}\bigr)
\times
\bigl(\Lambda_K,\phi_K,\phi_K^{-1}\bigr).
\]
Operações: Def.~\ref{def:somaprod}. Ordem: reticulado

Fechamento de $\otimes$, $\otimes^{*}$; reversibilidade de $\phi_K$;

\end{proposicao}

$d_k=2^{k+1}$,

$\gcd(2^{k+1},3^{K_k})=1$ $\Rightarrow$

\end{corolario}

\section{Catálogo das leis --- anel de índices}\label{sec:catalogo}

O ciclo de oito pertence ao \textbf{catálogo}, não à dimensão da torre.
A base das oito leis é o \emph{Corpo estelar} (\code{sub:oitoleis});
aqui só se verifica que o corpo de Peano as carrega e as induz.
\[
\mathcal L:=\mathbb{Z}/8\mathbb{Z},
\qquad
\boxed{\text{índice da lei}\neq\text{dimensão da torre}.}
\]
\label{def:anel}
Lei~0 é o índice $0$, não ``dimensão~$0$'' ($d_0=2$).

Soma e produto no catálogo: $\ell_1\oplus\ell_2=\ell_1+\ell_2\pmod8$,
$\ell_1\otimes\ell_2=\ell_1\ell_2\pmod8$. Neutros: $0$ e $1$.
Lei~7 $=-1$. Não é corpo: $2\otimes4=0\pmod8$.

\begin{center}\small
\begin{tabular}{@{}clll@{}}
índice & nome & operador / realização & fecho \\
\hline
$0$ & divisão do zero & $(+1)\oplus(-1)=0$; $0^{\dagger}=\infty$ & nulo \\
$1$ & dual & $\nu\colon x\mapsto x^{\dagger}$ & $\nu^{2}=\mathrm{id}$ \\
$2$ & bidual & $K^{**}=K$; $T^{\dagger}=-T$ & período $2$ no dual \\
$3$ & trial & $\tau$ em $\{-1,0,+1\}$ & $\tau^{3}=\mathrm{id}$ \\
$4$ & tetral & $\rho_4(T)=T+T^{*}$ & $d\mapsto2d$ \\
$5$ & pental & $\rho_5(x)=ix$ & $i^{2}=-1$, $i^{4}=\mathrm{id}$ \\
$6$ & hexal & $\oplus=\otimes$ na interface & $\operatorname{lcm}(2,3)=6$ \\
$7$ & octonião dual & $\mathbb{H}\times\mathbb{H}^{*}$ & $\operatorname{Ind}^{8}=\mathrm{id}_{\mathcal L}$ \\
\end{tabular}
\end{center}

\begin{observacao}
Anel $=$ catálogo das leis. Reticulado $=$ viveiro
$(\mathbb{N}_{>0},\mid)$ com $\operatorname{lcm}$ idempotente.
Não se identificam.
\end{observacao}

\begin{proposicao}[índice e dimensão]\label{cor:unicidade}
$k\in\mathbb{N}$ e $\ell\in\mathbb{Z}/8\mathbb{Z}$ são variáveis
distintas. O modelo associa $\ell_k=k\bmod8$ e
\[
(d_k,\iota_k,C_k)
=\bigl(2^{k+1},\;6\cdot2^{\lfloor k/3\rfloor},\;
12+6\lfloor k/3\rfloor\bigr).
\]
$\ell_{k+8}=\ell_k$, mas $d_{k+8}=256\,d_k$: a torre cresce.
\end{proposicao}

\begin{definicao}[indução no catálogo]\label{def:ind}
$\operatorname{Ind}(L)=L\oplus L_1$,
$L_n=\operatorname{Ind}^{n}(L_0)=n\cdot L_1$ em $\mathcal L$.
$\operatorname{Ind}^{8}=\mathrm{id}_{\mathcal L}$.
\end{definicao}

\begin{proposicao}[estrutura: as oito leis reaparecem em cada andar]\label{thm:induz}
Em cada andar $A_k$ da torre ($T_{k+1}=T_k+T_k^{*}$, $\dim=2^{k+1}$):
\begin{enumerate}
\item o catálogo $\mathcal L$ reaparece: $\ell_k=k\bmod8$ e
$L_n=\operatorname{Ind}^{n}(L_0)$ com $\operatorname{Ind}^{8}=\mathrm{id}$;
\item as representações fecham no mesmo sítio
($\nu^{2}=\tau^{3}=i^{4}=\mathrm{id}$; $\rho_4$ dobra a dimensão;
hexal só onde $\chi(1)\chi(3)=\chi(6)$);
\item o passo dos tecidos é o \emph{mesmo} em todos os andares --- a
estrela como construtor --- logo
$\mathcal L\to\mathcal L(\mathcal L)\to\cdots$
(\emph{Corpo estelar}): \textbf{não há Lei~$8$}, há a Lei~$0$ um nível
acima.
\end{enumerate}
\[
\boxed{
\text{oito leis}\ =\ \text{ciclo gerador};\quad
\text{torre}\ =\ \text{indução que as reaplica}.
}
\]
\end{proposicao}

\begin{observacao}[não é teorema externo]
Isto é \textbf{estrutura do modelo}, não um teorema demonstrado a
partir de axiomas mais fracos: o catálogo $\mathcal L$ é definido com
$\operatorname{Ind}^{8}=\mathrm{id}$, e a torre associa
$\ell_k=k\bmod8$ a cada andar. Distinguir:
$k$ = andar; $\ell_k=k\bmod8$ = índice cíclico; $d_k=2^{k+1}$ =
dimensão. A periodicidade de $\ell_k$ \textbf{não} é periodicidade de
$d_k$ ($d_{k+8}=256\,d_k$).
\end{observacao}

\section{Onde entram os milénios --- leituras, não base}\label{sec:milenio-leitura}

Os problemas do milénio \textbf{não fundam} este corpo.
A cadeia é
\[\text{leis}\to\text{corpo}\to\text{torre}\to\text{realizações}\to\text{leitura dos milénios}.\]
Nunca o inverso. Cada milénio é uma \emph{interpretação / derivação}
da lei correspondente (paper \emph{Milénio}): reformular o enunciado
cristalizado sobre o gerador vivo.
A tabela abaixo diz \emph{onde} cada um entra; a verdade da linha é a
da lei, ainda que o prémio não existisse.

\begin{center}\small
\begin{tabular}{@{}clll@{}}
lei & operador (base) & milénio (leitura) & chão clássico \\
\hline
$0$ & $(+1)\oplus(-1)$; $0^{\dagger}=\infty$ & --- (dois nulos) & divisão do zero \\
$1$ & $\nu\circ\nu=\mathrm{id}$ & Poincaré & dualidade $H^{k}\!\leftrightarrow\!H_{n-k}$ \\
$2$ & $K^{**}=K$; rotor & Yang--Mills; $P$ vs $NP$ & parada / estatuto \\
$3$ & trial $\{-1,0,+1\}$ & Riemann ($-1$); BSD ($+1$); Hodge ($0$) & Kronecker; Dirichlet; Lefschetz \\
$4$ & $T+T^{*}$, $|\det|=1$ & Navier--Stokes & Liouville (metade conservativa) \\
$5$ & $x^{2}=-1$ (bit $i$) & --- (sem milénio próprio) & $\mathbb{Z}[i]$ \\
$6$ & $\oplus=\otimes$ (interface) & --- (costura) & $\operatorname{lcm}(2,3)=6$ \\
$7$ & $\mathbb{H}\times\mathbb{H}^{*}$ & --- (topo / norma) & Hurwitz \\
\end{tabular}
\end{center}

\noindent No corpo de Peano, a entrada concreta é:
\begin{itemize}
\item Lei~1 / $a=-1$: estaca na borda; leitura Poincaré.
\item Lei~2 / $a=+1$: $\pm ix$; no centro $0$, leitura YM e $P$ vs $NP$.
\item Lei~3: trial e zeta / Pisot (Riemann); órbita metálica (BSD);
diagonal / tecidos (Hodge).
\item Lei~4: indução $d\mapsto2d$ e $\lvert\det A_m\rvert=1$ (NS
conservativo).
\item Leis~5--6: bit $i$ e interface hexal --- sem milénio; costuram.
\item Leis~0 e~7: nulos do catálogo; a torre $d_k$ continua.
\end{itemize}

\section{Música e Partitura}\label{sec:musica}

A música entrou como \textbf{linguagem de unificação}, não como
decoração. Nomes inventados pediam ao leitor montar a arquitectura
sozinho; \emph{metrónomo}, \emph{maestro}, \emph{batuta},
\emph{partitura}, \emph{orquestra} já trazem \textbf{relações
estruturais embutidas}:
\[
\text{partitura}
\;\rightarrow\;
\text{maestro}
\;\rightarrow\;
\text{batuta}
\;\rightarrow\;
\text{instrumentos}
\;\rightarrow\;
\text{orquestra}.
\]
O leitor vê de imediato especificação, interface de execução, canal de
comando, realizadores e composição --- e isso traduz a arquitectura
matemática:
\[
\Pi
\;\rightarrow\;
\text{Maestro}
\;\rightarrow\;
\pi_k
\;\rightarrow\;
G_i
\;\rightarrow\;
\mathcal{O}.
\]
A vantagem: os nomes \textbf{não precisam explicar} a arquitectura
inteira --- a semântica já carrega a relação. A roupa musical
\textbf{não substitui} a matemática: \textbf{revela} a arquitectura
que a matemática já tinha.
\[
\boxed{
\text{A música não foi colocada sobre o modelo:}
\quad
\text{foi encontrada como a linguagem em que o modelo já sabia falar.}
}
\]
Níveis que a música separa naturalmente --- os mesmos que a matemática
já distinguia:
\begin{center}\small
\begin{tabular}{@{}lll@{}}
nível & objecto musical & função \\
\hline
estrutura & corpo / conservatório & o que se conserva \\
especificação & partitura ($\Pi$) & o quê \\
tempo & metrónomo & quando (lê) \\
projecção & maestro & projecta $\pi_k$ \\
canal & batuta & canaliza $I$ \\
realização & instrumento ($G$) & naipe / gerador \\
composição & orquestra ($\mathcal{O}$) & fusão sob o maestro \\
atestação & métrica (Lyapunov dualizado) & atesta $\lambda^{+}{+}\lambda^{-}=0$ \\
\end{tabular}
\end{center}
Consequência operacional dos Teoremas do Maestro e do Metrónomo
(\S\ref{sec:maestro-thm}): a teoria constrói a retração; o maestro
projecta; o metrónomo lê; a música é o resultado observado.
\[
\boxed{
\text{Conservatório}
\;\to\;
\text{Pera}
\;\to\;
\text{Batuta}
\;\to\;
\text{Ponta}
\;\to\;
\text{Música}.
}
\]
Cadeia tipográfica (não retroage às leis):
\[
\text{leis}\to\text{corpo/torre}\to
\text{Teorema Central}
\to\pi_k\to
(\text{Maestro},\,\text{Metrónomo})
\to\text{música}/\text{partitura}/\text{naipes}/\text{orquestra}.
\]
Música $=$ realização operacional de $\pi_k$ no tempo; partitura $=$
codificação ($\Pi$); naipes $=$ classificação de $G$; orquestra $=$
composição sob o maestro.

\begin{definicao}[Música]\label{def:musica}
Uma \textbf{música} é um triplo
\[
\mathcal{M}\;=\;(d,\,S,\,G)
\]
onde
\begin{itemize}
\item $d\in\mathbb{N}_{>0}$ é a \textbf{dimensão} (grau / andar da torre
$d_k=2^{k+1}$, ou o grau $n$ do corpo do catálogo);
\item $S$ é o conjunto de \textbf{sementes} --- as condições iniciais da
órbita (tipicamente o par dual da divisão do zero $\{0,\infty\}$, ou a
semente da estrela-espiral que declara a geometria);
\item $G$ é o \textbf{gerador} --- o \textbf{instrumento musical}: o
operador que move o corpo. \textbf{Restrição de classe} deste modelo:
músicas com $G$ no sector antissimétrico $G^{\dagger}=-G$ (Lei~2).
Não se afirma que todo instrumento físico real satisfaz
$G^{\dagger}=-G$ --- só os da classe aqui definida.
\end{itemize}
\[
\boxed{\text{convenção: cada corpo do catálogo realiza-se como uma música}.}
\]
O catálogo indexa; a torre dá dimensão; $S$ escolhe órbita; $G$ toca.
Isto não acrescenta axioma às leis.
\end{definicao}

\begin{proposicao}[duas músicas]\label{prop:duas-musicas}
Fixados $d$ e $G$, sementes distintas $S\neq S'$ dão músicas distintas
(órbitas distintas sob o mesmo instrumento). Fixada a semente, geradores
não conjugados dão músicas distintas. A dimensão separa andares: se
$d\neq d'$, as músicas não coincidem.
\end{proposicao}

\begin{definicao}[Partitura]\label{def:partitura}
A \textbf{partitura} transporta a música em dois blocos:
\[
\Pi
\;=\;
\Pi_{\mathrm{assinatura}}
\;+\;
\Pi_{\mathrm{notação}},
\]
\[
\Pi_{\mathrm{assinatura}}
\;=\;
(d,\,S,\,G,\,\ell,\,C,\,\Delta),
\qquad
\Pi_{\mathrm{notação}}
\;=\;
(P,\,\kappa,\,\alpha,\,\tfrac{n}{m},\,F,\,R,\,B,\,\omega),
\]
com $P$ pentagrama, $\kappa$ clave, $\alpha$ armadura, $\tfrac{n}{m}$
compás, $F$ figuras, $R$ silêncios, $B$ barras, $\omega$ final.
A assinatura classifica; a notação escreve. Nenhum bloco funda as leis.
\[
\boxed{
\text{música}=\text{realização};\quad
\text{partitura}=\text{assinatura}+\text{notação}.
}
\]
\end{definicao}

\begin{proposicao}[ida e volta relativa]\label{prop:partitura-id}
Relativamente ao catálogo $\mathcal L$ e às regras de decodificação
declaradas, ler e tocar é involutivo na classe: $\Pi$ determina
$\mathcal{M}$ até conjugação de $G$, e $\mathcal{M}$ determina
$\Pi_{\mathrm{assinatura}}$ (\code{tests/assinatura.c}).
Autocontida \emph{relativamente} a esse catálogo e decodificador ---
não em absoluto sem regras. $\Pi$ cavalga o corpo.
\end{proposicao}

\begin{observacao}
Música $\neq$ partitura. A música \emph{soa} (dimensão, sementes,
instrumento em acto). A partitura \emph{codifica} (assinatura finita,
autocontida). O Peano é o piano; o catálogo é o reportório; cada entrada
é uma peça; a partitura é a folha.
A realização gráfica vive no tradutor WASM:
\code{partitura/emite.c} emite \code{papers/partitura\_vozes.tex};
\code{papers/partitura.tex} desenha; \code{tests/tex} / \code{tex.wasm}
compõem \code{partitura-wasm.pdf}. Cadeia:
relógio$\to$metrónomo$\to$maestro$\to$batuta$\to\Pi\to\pi_k\to$orquestra.
A partitura não inventa informação fora da assinatura.
A figura~\ref{fig:partitura-sistema} resume as vozes.
\end{observacao}

\subsection*{O alfabeto da partitura}

\begin{definicao}[Pentagrama musical]\label{def:pentagrama}
O \textbf{pentagrama} é o suporte geométrico onde a partitura se escreve:
cinco linhas e quatro espaços. No modelo, é a
\textbf{realização particular} (não identidade estrutural) que associa
o suporte escrito à Lei~5 (pental / bit $i$) e, na geometria do maestro,
ao \textbf{cone} / \textbf{pera} (Def.~\ref{def:pera}):
\[
\Pi_{\mathrm{penta}}\;:=\;5\text{ linhas}
\quad\text{(convenção de realização)},
\qquad
\boxed{\text{pentagrama}=\text{pera}/\text{cone}}.
\]
Não se prova pela Lei~5 que o pentagrama clássico tenha cinco linhas;
escolhe-se a realização.
É a régua do andar vista de frente: a altura (linha/espaço) endereça a
altura sonora no cone; o eixo horizontal endereça o tempo (Metrónomo).
A música que se escreve nele é a \textbf{espiral} da vareta
(Def.~\ref{def:batuta-linear}) a projectar a orquestra --- não a
diagonal. Sem pentagrama não há sítio para as figuras --- como sem
$C_k$ não há janela na torre.
\end{definicao}

\begin{definicao}[Clave]\label{def:clave}
Uma \textbf{clave} é a âncora de leitura no pentagrama: fixa qual linha
é a referência e, com ela, o isomorfismo entre posição e altura.
\emph{Realização / interpretação:} escolha de referencial no
pentagrama, lida como base dual --- duas leituras (aguda / grave)
trocadas pela estaca $\nu$ (Lei~1):
\[
\mathrm{clave}_{\mathrm{sol}}
\;\overset{\nu}{\longleftrightarrow}\;
\mathrm{clave}_{\mathrm{fá}}.
\]
A clave não cria notas: escolhe o referencial. Instrumento distinto
pode exigir clave distinta; a música $(d,S,G)$ declara-a na partitura.
\end{definicao}

\begin{definicao}[Figuras musicais]\label{def:figuras}
As \textbf{figuras} são as durações sonoras activas --- potências de
$1/2$ a partir da unidade (semibreve $=1$, mínima $=\tfrac12$,
semínima $=\tfrac14$, \ldots):
\[
f_k\;=\;2^{-k},\qquad k\geq0.
\]
\emph{Realização:} potências de $1/2$ espelham a torre binária no
tempo (analogia operacional com $d\mapsto2d$, não teorema de
identidade).
Uma figura ocupa posição no pentagrama (altura) e intervalo no
compás (duração). Soa: medida não nula sob o instrumento $G$.
\end{definicao}

\begin{definicao}[Silêncios musicais]\label{def:silencios}
Os \textbf{silêncios} são o dual das figuras: mesma duração, medida
nula (não soa). A estaca troca
\[
\mathrm{figura}\;\overset{\nu}{\longleftrightarrow}\;\mathrm{silêncio}
\]
com $\nu\circ\nu=\mathrm{id}$ --- \emph{realização} da Lei~1 no
alfabeto rítmico. O silêncio de compás inteiro \emph{interpreta}
localmente a Lei~0. Sem silêncios a partitura não respira; sem figuras
não há música.
\end{definicao}

\begin{definicao}[Armadura]\label{def:armadura}
A \textbf{armadura} é o conjunto fixo de alterações (sustenidos /
bemóis) à cabeça da partitura: escolhe a \textbf{órbita tonal} --- a
semente metálica / o sinal de $\Delta$ --- sem a reescrever em cada
figura.
\[
\mathrm{Armadura}\;=\;\text{semente tonal de }\mathcal{M}
\;\subset\;S.
\]
É parte da assinatura: mesma dimensão e instrumento, armaduras
distintas $\Rightarrow$ músicas distintas (Proposição~\ref{prop:duas-musicas}).
\end{definicao}

\begin{definicao}[Indicador de compás]\label{def:compas}
O \textbf{indicador de compás} é o par
\[
\frac{n}{m}
\]
à cabeça da partitura: $n$ pulsos por compás, $m$ a figura que vale um
pulso. Valores deste modelo: períodos fundamentais $\{2,3,4,8\}$ e
interfaces derivadas $\{6,12,24\}$:
\[
m,n\in\{2,3,4,6,8,12,24\}.
\]
O compás é a janela rítmica (realização análoga a $C_k$). Hexal
($6=\operatorname{lcm}(2,3)$) é interface, não período fundamental.
\end{definicao}

\begin{definicao}[Barra de compás]\label{def:barra}
A \textbf{barra de compás} é a partição vertical que fecha um
compás e abre o seguinte: \emph{realização} do corte no eixo do tempo
(analogia com max-cut / Lei~0 local --- não identidade).
\[
|\;=\;\text{fronteira de medida};\qquad
\|\;=\;\text{barra dupla (secção)}.
\]
Não dura: marca. Sem barras o indicador de compás não se materializa;
com elas a partitura fica um reticulado temporal (viveiro no tempo).
\end{definicao}

\begin{definicao}[Final da partitura]\label{def:final}
O \textbf{final da partitura} é o fecho autocontido: barra final
(barra fina + barra grossa) --- a peça termina com resíduo $0$.
\emph{Realização do fecho:} declara-se $\omega$ tal que a leitura
termina com resíduo $0$ na classe (Proposição~\ref{prop:partitura-id}).
A analogia com $\operatorname{Ind}^{8}=\mathrm{id}_{\mathcal L}$ e
$\nu\circ\nu=\mathrm{id}$ é interpretativa: o final da peça
\emph{usa} esses fechos, não os prova.
\end{definicao}

\begin{proposicao}[a partitura completa]\label{prop:partitura-completa}
Uma partitura $\Pi$ declara, nesta ordem operacional:
\begin{enumerate}
\item pentagrama (suporte);
\item clave (referencial);
\item armadura (semente tonal);
\item indicador de compás (janela rítmica);
\item figuras e silêncios (alfabeto, duais);
\item barras de compás (partições);
\item final (fecho, resíduo $0$).
\end{enumerate}
Estes campos entram em $\Pi_{\mathrm{notação}}$
(Def.~\ref{def:partitura}). Sem eles a notação está incompleta;
a assinatura sozinha classifica mas não escreve a peça. Música $=$
realização; partitura $=$ transporte --- nenhum funda as leis.
\end{proposicao}

\begin{proposicao}[$\Pi$ na cadeia operacional e na UI]\label{prop:pi-cadeia}
A partitura $\Pi$ (Def.~\ref{def:partitura}) entra em três sítios sem
os fundir:
\begin{enumerate}
\item \textbf{Maestro.}
$\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$
(Teorema~\ref{thm:proj-maestro}) --- $\Pi$ determina o projectar.
\item \textbf{Histerese.} $H_{k+1}=\mathrm{Maestro}(\Pi,H_k)$
(Teorema~\ref{thm:histerese}) --- $\Pi$ orienta a memória da borda.
\item \textbf{Assistente / rede dual.} Na UI,
\code{partituraFala} extrai $\Pi$ da fala; o controlo
$\mathcal{F}$ escolhe o ramo; o Metrónomo lê o resíduo $r$
relativamente a $\Pi$ (\code{app/src/maestro.js},
\code{app/src/assistente.js}; Dual Sort
\code{papers/dualsort.tex}).
\end{enumerate}
\[
\boxed{
\Pi
\;=\;
\text{o quê};
\qquad
I
\;=\;
\text{o sentido};
\qquad
\mathcal{F}
\;=\;
\text{o ramo};
\qquad
r
\;=\;
\text{a atestação}.
}
\]
A realização gráfica (\code{papers/partitura.tex},
\code{partitura-wasm.pdf}) desenha $\Pi$; não substitui o Teorema
Central nem a rede dual.
\end{proposicao}

\begin{definicao}[Página --- $\pi_{\mathrm{página}}$]\label{def:pi-pagina}
A \textbf{página} é um andar de composição sob o mesmo Maestro: dimensão
do relógio, não interrupção externa.
\[
\boxed{
\mathrm{tick}
\to
\mathrm{compasso}
\to
\mathrm{partitura}
\to
\mathrm{página}
\to
\mathrm{torre}
}
\]
$\pi_{\mathrm{página}}$ projecta o bloco contínuo no suporte $P_n$; se não
cabe, $P_n\xrightarrow{\pi}P_{n+1}$ --- muda o suporte, não o estado
(relógio, espiral, voz, métrica, $\Pi$, linha dourada). Realização no
tradutor: \code{\textbackslash pipagina}.
\end{definicao}

\subsection*{Os quatro naipes --- Hurwitz no instrumento}

Hurwitz classifica as álgebras reais de composição com norma
euclidiana multiplicativa: graus (dimensões da álgebra)
$1,2,4,8$ --- $\mathbb{R},\mathbb{C},\mathbb{H},\mathbb{O}$
(\emph{Corpo estelar}). Os \textbf{naipes} são uma
\textbf{definição de realização} deste modelo: nomes musicais
atribuídos a essas quatro classes do gerador $G$. Não é um teorema de
Hurwitz que ``Cordas$=\mathbb{R}$''; é convenção do modelo. Não há
quinto naipe porque não há quinta álgebra de composição.

\begin{definicao}[Naipe]\label{def:naipe}
Um \textbf{naipe} é uma classe de instrumentos $G$ indexada pelo
\emph{grau de composição} (dimensão da álgebra de Hurwitz), não pela
dimensão $d_k$ da torre nem pelo índice $\ell$ da lei:
\[
\mathrm{Naipe}\;\in\;\{\text{Cordas},\,\text{Madeiras},\,\text{Metais},\,\text{Percussão}\}.
\]
Cada música $\mathcal{M}=(d,S,G)$ declara o naipe do seu $G$. Distinguir:
grau de composição $\in\{1,2,4,8\}$; dimensão real do espaço de estados;
classe do gerador; índice de lei. A orquestra funde naipes por estrela.
\[
\boxed{\text{naipe}\;=\;\text{realização nomeada do grau Hurwitz}.}
\]
\end{definicao}

\begin{tabular}{@{}cllll@{}}
naipe & grau comp. & álgebra & o que perde na dobra & carácter do $G$ \\
\hline
\textbf{Cordas} & $1$ & $\mathbb{R}$ & --- (ordenado) & contínuo, ordem total, arco \\
\textbf{Madeiras} & $2$ & $\mathbb{C}$ & a ordem & sopro, fase $i$, período $4$ \\
\textbf{Metais} & $4$ & $\mathbb{H}$ & a comutação & família metálica $A_m$, bocal \\
\textbf{Percussão} & $8$ & $\mathbb{O}$ & a associatividade & golpe discreto, Hurwitz conta \\

\begin{definicao}[Cordas]\label{def:cordas}
O naipe das \textbf{Cordas} \emph{realiza} (definição) o corpo real ($\mathrm{grau\ comp.}=1$): o
instrumento vibra em contínuo com \textbf{ordem} --- altura total,
arco / corda. $G$ preserva a ordem da régua. É a base da torre
($A_0\sim\mathbb{R}$): sem cordas não há primeiro andar.
\end{definicao}

\begin{definicao}[Madeiras]\label{def:madeiras}
O naipe das \textbf{Madeiras} \emph{realiza} o corpo complexo ($\mathrm{grau\ comp.}=2$): o
sopro introduz a \textbf{fase} --- bit $i$, Lei~5 no tubo --- e perde a
ordem real. $G\cong aI+bJ$ com $J^{\dagger}=-J$. A dobra
$\mathbb{R}\to\mathbb{C}$ é o primeiro tecido: corda ordenada torna-se
coluna de ar com período $4$.
\end{definicao}

\begin{definicao}[Metais]\label{def:metais}
O naipe dos \textbf{Metais} \emph{realiza} o corpo quaternião ($\mathrm{grau\ comp.}=4$): a
\textbf{família metálica} $A_m$ realizada no bocal --- norma
$|N|=1$, $\det=-1$, não comuta. $G$ vive em $\mathbb{H}$; a dobra
$\mathbb{C}\to\mathbb{H}$ perde a comutação. O nome coincide com a
teoria: metais $=$ metálicas no instrumento.
\end{definicao}

\begin{definicao}[Percussão]\label{def:percussao}
O naipe da \textbf{Percussão} \emph{realiza} o corpo octonião ($\mathrm{grau\ comp.}=8$): o
golpe discreto --- \textbf{contar} (Hurwitz) sem associar. $G$ fecha a
norma de composição; a dobra $\mathbb{H}\to\mathbb{O}$ perde a
associatividade. É o topo da leitura pela norma (Lei~7); a estrela
recomeça acima, mas o naipe pára em oito.
\end{definicao}

\begin{proposicao}[não há quinto naipe]\label{prop:quatro-naipes}
A lista $\{\text{Cordas},\,\text{Madeiras},\,\text{Metais},\,\text{Percussão}\}$
é exacta e esgotada: graus $1,2,4,8$. Fundir naipes é a estrela entre
corpos de composição; o resultado sobe de grau na torre, sem inventar
naipe novo. O indicador de compás e a armadura são ortogonais ao naipe
(ritmo e tonalidade); o naipe classifica só o $G$.
\end{proposicao}

\subsection*{Maestro e orquestra}

O instrumento toca. O \textbf{maestro} \textbf{não} é entidade
matemática necessária ao corpo operacional: é
\textbf{interface da realização musical}
que casa tempo e dinâmica entre naipes --- estruturalmente
$(\mathrm{relógio},\,\mathrm{inversor},\,\Pi)$; musicalmente
$(\mathrm{Metr\acute{o}nomo},\,\mathrm{Batuta},\,\Pi)$.
Sem música, o corpo não precisa de maestro.

\begin{definicao}[Maestro]\label{def:maestro}
O \textbf{maestro} é a interface da realização musical. Na linguagem
estrutural do corpo:
\[
\mathrm{Maestro}
\;=\;
\bigl(\underbrace{\text{relógio}}_{\text{Lei~6 / hexal}},\;
\underbrace{\text{inversor}}_{\text{trial }\{-1,0,+1\}},\;
\Pi\bigr).
\]
Na linguagem musical da realização:
\[
\boxed{
\mathrm{Maestro}
\;=\;
(\mathrm{Metr\acute{o}nomo},\,\mathrm{Batuta},\,\Pi).
}
\]
\begin{itemize}
\item Relógio $\to$ pulso; inversor $\to$ direcção; $\Pi\to$ conteúdo.
\item O relógio (Lei~6) ancora compás e barras; meia volta é a
velocidade máxima (\emph{Corpo estelar}).
\item O inversor é o operador $I\colon\{-1,0,+1\}$ --- retrair, neutro,
empurrar --- sem buffer (\code{tests/inversor\_trial.c}).
\end{itemize}
O maestro $\notin\{\text{naipes}\}$ e não é necessário ao corpo
operacional --- só à realização musical.
\end{definicao}

\begin{definicao}[Metrónomo]\label{def:metronomo}
O \textbf{relógio} é a estrutura temporal abstrata do maestro.
O \textbf{metrónomo} é a sua \textbf{realização musical}:
\[
\boxed{
\mathrm{Metr\acute{o}nomo}
\;=\;
\mathrm{Rel\acute{o}gio}_{\mathrm{musical}}.
}
\]
Fornece o pulso e ancora o indicador de compás e as barras.
Como \textbf{régua} (leitura), o Teorema~\ref{thm:metronomo} eleva-o
a realização métrica dual da dinâmica do maestro --- sem o confundir
com a batuta (canal) nem com o inversor ($I$).
Correspondência: $\mathrm{Metr\acute{o}nomo}\leftrightarrow\mathrm{Rel\acute{o}gio}$.
\end{definicao}

\begin{definicao}[Batuta]\label{def:batuta}
O \textbf{inversor} é o operador abstrato $I\colon\{-1,0,+1\}$ com
$-1=$ retrair, $0=$ neutro, $+1=$ empurrar.
A \textbf{batuta} \textbf{não} é o inversor: é o
\textbf{canal / meio físico} que o realiza e o transmite aos músicos:
\[
\boxed{
\mathrm{Batuta}
\;=\;
\text{realização}(I)
\;=\;
\text{canal do inversor}.
}
\]
O mesmo inversor pode realizar-se sem batuta --- gesto da mão,
regência corporal, interface digital. A batuta é \emph{uma} realização
possível, não a identidade $I$. Correspondência:
$\mathrm{Batuta}\leftrightarrow\mathrm{Inversor}$ (transmite; não
substitui).
\end{definicao}

\begin{proposicao}[cadeia do maestro]\label{prop:batuta-metro}
\[
\boxed{
\mathrm{Rel\acute{o}gio}_{\mathrm{lei/tempo}}
\;\rightarrow\;
\mathrm{Metr\acute{o}nomo}_{\mathrm{pulso}}
}
\qquad
\boxed{
\mathrm{Inversor}_{\mathrm{trial}}
\;\rightarrow\;
\mathrm{Batuta}_{\mathrm{gesto}}
}
\]
\[
\boxed{
(\mathrm{Metr\acute{o}nomo},\,\mathrm{Batuta},\,\Pi)
\;\rightarrow\;
\mathrm{Maestro}
\;\rightarrow\;
\mathrm{Orquestra}.
}
\]
Fechos intactos: $\operatorname{lcm}(2,3)=6$ (relógio/hexal);
$\tau^{3}=\mathrm{id}$ (inversor/trial). A batuta não altera $I$; só o
canaliza.
\end{proposicao}

\begin{proposicao}[fecho operacional do maestro]\label{prop:maestro-fecha}
\textbf{Se} o maestro partilha um único metrónomo (relógio, Lei~6) e um
único canal de inversor (batuta ou outro meio) por todos os $G_i$, com
a mesma $\Pi$, \textbf{então} a fusão opera com resíduo $0$ no sentido
do casamento de relógios (\emph{Corpo estelar}). Hipótese operacional,
não teorema independente. Sem tempo/direcção comuns, cada naipe desafina.
\end{proposicao}

\begin{definicao}[Orquestra]\label{def:orquestra}
Uma \textbf{orquestra} $\mathcal{O}$ é a fusão por estrela de um
conjunto de naipes (instrumentos $G$ classificados), sob um maestro:
\[
\mathcal{O}
\;=\;
\mathrm{Estrela}\bigl(\{G_i\},\;\mathrm{Maestro}\bigr).
\]
Os naipes não se colam num produto interno único: a estrela liga sem
fundir (\emph{Catálogo}). A espécie da orquestra é o \textbf{suporte}
em graus de Hurwitz que ela realiza.
\end{definicao}

\begin{definicao}[Orquestra sinfónica / filarmónica]\label{def:sinfonica}
\textbf{Convenção deste modelo:} sinfónica e filarmónica recebem a
\emph{mesma} realização operacional --- suporte completo dos quatro
graus de composição:
\[
\mathrm{supp}(\mathcal{O}_{\mathrm{sinf}})
\;=\;
\mathrm{supp}(\mathcal{O}_{\mathrm{fil}})
\;=\;
\{1,2,4,8\}.
\]
Não se afirma como facto universal da música histórica que sejam o
mesmo objecto; aqui igualam-se por definição de realização. Os nomes
permanecem distintos na linguagem; o suporte Hurwitz coincide.
\end{definicao}

\begin{definicao}[Orquestra de câmara]\label{def:camara}
A \textbf{orquestra de câmara} é a fusão \textbf{reduzida}: suporte
próprio em graus de Hurwitz, mas cardinalidade e interface menores
--- viveiro com $\operatorname{lcm}$ curto, janela $C$ de câmara.
\[
\mathrm{supp}(\mathcal{O}_{\mathrm{câm}})
\;\subseteq\;
\{1,2,4,8\},
\qquad
|\mathcal{O}_{\mathrm{câm}}|
\;\ll\;
|\mathcal{O}_{\mathrm{sinf}}|.
\]
Pode usar vários naipes em miniatura; o que a distingue é a
\textbf{escala} (câmara $=$ interface pequena), não a proibição de um
naipe. O maestro continua: relógio + inversor em $\Pi$, com menos
vozes a casar.
\end{definicao}

\begin{definicao}[Orquestra de cordas]\label{def:orq-cordas}
A \textbf{orquestra de cordas} é a orquestra cujo suporte é
\textbf{só} o naipe das Cordas:
\[
\mathrm{supp}(\mathcal{O}_{\mathrm{cord}})
\;=\;
\{1\}
\;=\;
\{\text{Cordas}\}.
\]
Grau $1$ ($\mathbb{R}$, ordem): arco e altura total, sem madeiras,
metais nem percussão. O maestro reduz-se ao relógio + inversor sobre
um único corpo de composição --- ainda interface, nunca quinto naipe.
\end{definicao}

\begin{proposicao}[ordem por suporte e escala]\label{prop:orq-hier}
Não se afirma inclusão de conjuntos de músicos. Define-se a relação
$\preceq$ por suporte Hurwitz e escala (cardinalidade / $\operatorname{lcm}$
da interface):
\[
\mathrm{supp}(\mathcal{O}_{\mathrm{cord}})=\{1\}
\;\subseteq\;
\mathrm{supp}(\mathcal{O}_{\mathrm{câm}})
\;\subseteq\;
\{1,2,4,8\}
\;=\;
\mathrm{supp}(\mathcal{O}_{\mathrm{sinf}})
\;=\;
\mathrm{supp}(\mathcal{O}_{\mathrm{fil}}),
\]
e tipicamente $|\mathcal{O}_{\mathrm{câm}}|\ll|\mathcal{O}_{\mathrm{sinf}}|$.
Uma orquestra de câmara \textbf{não} é automaticamente subestrutura
de uma sinfónica: só o suporte/escala se comparam. Toda orquestra leva
maestro.
\[
\boxed{
\text{ordem}\;=\;\text{suporte/escala},\ \text{não}\ \subset\ \text{de elenco}.
}
\]
\end{proposicao}

\subsection*{Realização musical --- Conservatório, Pera, Batuta}

Consequência do Teorema~\ref{thm:proj-maestro}: a geometria do canal
que transporta $\pi_k$.
\[
\boxed{
\mathrm{maestro}
\;=\;
\text{realização dinâmica de }\pi_k.
}
\]

\begin{observacao}[estrutura de projecção do maestro]\label{obs:proj-dinamica}
Papéis na realização (não criam $\pi_k$ --- só o executam):
metrónomo $=$ \emph{quando}; batuta $=$ \emph{como}
(pera + vareta + ponta); partitura $\Pi=$ \emph{o quê}; orquestra $=$
\emph{onde}; conservatório $=$ \emph{acima}.
\[
\boxed{
\mathrm{Metr\acute{o}nomo}
\;\longrightarrow\;
\mathrm{Maestro}(\pi_k)
\;\longrightarrow\;
\mathrm{Orquestra}.
}
\]
\end{observacao}

\begin{definicao}[Conservatório]\label{def:conservatorio}
O \textbf{conservatório} é o andar / espaço \textbf{acima} da realização
musical: o sítio onde mora a estrutura a projectar --- conservação da
norma de Gentil e/ou o cone de direcções admissíveis (espaço onde a
distância própria nula não entorna). Não é a orquestra nem a música:
é o \emph{depósito conservado} do qual o maestro desce a realização.
\[
\boxed{
\mathrm{Conservat\acute{o}rio}
\;=\;
\text{estrutura superior conservada}
\;=\;
\bigl(\text{norma Gentil},\;\text{cone}\bigr)_{\mathrm{acima}}.
}
\]
Estatuto: definição / interpretação da realização --- não lei nova.
A música abaixo é a sombra; o conservatório é o objecto que a projecção
não inventa.
\end{definicao}

\begin{definicao}[Pera]\label{def:pera}
A \textbf{pera} é a \textbf{base} da batuta: o cabo onde a mão assenta
e onde o cone do conservatório \textbf{entra} no canal. Operacionalmente
\[
\boxed{
\mathrm{Pera}
\;=\;
\text{base da batuta}
\;=\;
\text{cone acoplado em }s=0.
}
\]
Não é a vareta (linear em $s$) nem a ponta (onde a trajectória se
escreve): é o encaixe
$\mathrm{Conservat\acute{o}rio}\hookrightarrow\mathrm{Batuta}$.
Correspondência:
$\mathrm{Pera}\leftrightarrow\mathrm{Cone}_{\mathrm{base}}$.
\end{definicao}

\begin{definicao}[Batuta linear]\label{def:batuta-linear}
A batuta, como canal geométrico, tem três peças:
\begin{itemize}
\item \textbf{pera} (base, $s=0$) --- cone / cabo (Def.~\ref{def:pera});
\item \textbf{vareta} --- segmento linear
$B(s,t)=p(t)+s\,u(t)$ com $0<s\le L$;
\item \textbf{ponta} $P(t)=B(L,t)=p(t)+L\,u(t)$.
\end{itemize}
A geometria em $s$ na vareta é linear; a trajectória $P(t)$ não precisa
ser --- periódica, espiralada, reflectida. Cadeia operacional:
\[
\boxed{
\mathrm{Conservat\acute{o}rio}
\;\xrightarrow{\mathrm{Pera}}\;
\text{vareta linear}
\;\rightarrow\;
\text{ponta }P(t)
\;\rightarrow\;
\text{comando }I
\;\rightarrow\;
\text{som}.
}
\]
O inversor $I\in\{-1,0,+1\}$ continua o comando; a batuta só o
canaliza (Def.~\ref{def:batuta}). A pera não é $I$: é a base cónica
do canal.
\end{definicao}

\begin{proposicao}[rede dual no canal musical]\label{prop:rede-no-canal}
Na UI / assistente, o comando $I$ da batuta (Def.~\ref{def:batuta})
escolhe o ramo do controlador dual
(\S\ref{sec:rede-dual}): frente $F_P$, retenção na banda, ou volta
$F_H$. A geometria Conservatório$\to$Pera$\to$Batuta \textbf{não}
define $x'$; só canaliza $I$. Em troca, a rede dual \textbf{não}
substitui o conservatório: decide sob a atestação do Metrónomo
relativamente a $\Pi$ (Prop.~\ref{prop:pi-cadeia}).
\[
\boxed{
\Pi
\;+\;
\mathrm{Batuta}(I)
\;\longrightarrow\;
\mathcal{F}\colon(x,h)\mapsto(x',h')
\;\longrightarrow\;
\text{Metrónomo }(r,\,\lambda^{+}+\lambda^{-}=0).
}
\]
Fundamento: Teoremas~\ref{thm:proj-maestro},~\ref{thm:metronomo},
\ref{thm:rede-dual}; Prop.~\ref{prop:central-operacional},
\ref{prop:pi-cadeia}.
\end{proposicao}

\begin{proposicao}[cadência e carga]\label{prop:tick-carga}
O metrónomo fornece o parâmetro temporal $t$ (ticks $t_0,t_1,\ldots$
ou relógio contínuo). A partitura impõe o conteúdo; o maestro
transforma $(\Pi,t)$ em gesto na batuta; os $G_i$ realizam:
\[
\boxed{
\Pi+t
\;\longrightarrow\;
B(t)
\;\longrightarrow\;
G_i(t).
}
\]
Hipótese operacional da realização musical --- não axioma do corpo;
realiza $\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$
(Teorema~\ref{thm:proj-maestro}).
\end{proposicao}

\begin{proposicao}[espiral áurea como realização]\label{prop:espiral-real}
A cadeia
\[
\mathrm{Conservat\acute{o}rio}
\;\xrightarrow{\pi_k}\;
\text{órbita inferior}
\]
não introduz axioma. Se a órbita inferior é a metálica $m=1$
($\sigma^{2}=\sigma+1$, $\sigma=\varphi$), a espiral áurea aparece
como realização da dinâmica já definida --- sombra do cone na
pera projectada no plano musical --- não como lei nova.
\[
\boxed{
\mathrm{Conservat\acute{o}rio}
\;\xrightarrow[\Pi]{\mathrm{Pera}+\pi_k+\text{metrónomo}}\;
\text{espiral/música}
\;\longrightarrow\;
\text{orquestra}.
}
\]
\end{proposicao}

\begin{proposicao}[milénios lêem, não geram]\label{prop:milenio-dinamica}
Os milénios permanecem leituras das leis
(\S\ref{sec:milenio-leitura}). Em paralelo:
\[
\text{leis}\to\text{dinâmica}\to\text{realização musical},
\qquad
\text{leis}\to\text{milénios (interpretação)}.
\]
Um milénio pode \emph{descrever} uma dinâmica que a música realiza;
não a gera. A ponta da batuta é onde a estrutura superior se torna
visível como trajectória $\to$ som.
\end{proposicao}

\section{Teorema da Histerese --- memória da borda}\label{sec:histerese}

A histerese clássica introduz dois limiares $(\theta_{\mathrm{high}},\theta_{\mathrm{low}})$
para que \emph{entrar} e \emph{sair} não coincidam. No Corpo de Peano
\textbf{não} se postula uma Lei~$9$: os limiares são as duas faces da
volta já medida pelo Metrónomo; a ``zona cinzenta'' é a borda
(Def.~\ref{def:borda}); a memória é o estado do relógio $X_k$.

\subsection*{Derivação do conceito clássico}

No modelo clássico de Schmitt / histerese de limiar:
\[
y_{k+1}
=
\begin{cases}
+1 & \text{se }x_k\geq\theta_{\mathrm{high}},\\
-1 & \text{se }x_k\leq\theta_{\mathrm{low}},\\
y_k & \text{caso contrário (zona cinzenta)}.
\end{cases}
\]
A zona cinzenta é precisamente o lugar onde a decisão ainda não
atravessou; no nosso vocabulário:
\[
\boxed{
A
\;|\;
\text{borda }B
\;|\;
A^{c}.
}
\]
Os dois limiares não são cortes arbitrários independentes: sob a
leitura dual do Metrónomo (Teorema~\ref{thm:metronomo}),
\[
\boxed{
\theta_{\mathrm{high}}\;\leftrightarrow\;\lambda^{+}
\quad\text{(expansão)},
\qquad
\theta_{\mathrm{low}}\;\leftrightarrow\;\lambda^{-}
\quad\text{(retração)},
}
\]
com o fecho da volta
\[
\boxed{\lambda^{+} + \lambda^{-} = 0}.
\]
Assim $\mathrm{entrar}\neq\mathrm{sair}$ deixa de ser axioma extra e
passa a ser corolário da \textbf{borda orientada} pela batuta
$I\in\{-1,0,+1\}$ (Lei~3).

\begin{definicao}[estado de histerese]\label{def:H-k}
Seja $X_k$ o suporte activo no tick $k$, $B_k$ a borda entre $X_k$ e o
seu complemento, e $I_k\in\{-1,0,+1\}$ a direcção da batuta. O
\textbf{estado de histerese} é
\[
\boxed{
H_k = \bigl(X_k,\,B_k,\,I_k\bigr).
}
\]
\end{definicao}

\begin{teorema}[Histerese --- composição das leis existentes]\label{thm:histerese}
Seja $\Pi$ a partitura e $\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$
a realização do Maestro (Teorema~\ref{thm:proj-maestro}). Então
\[
\boxed{
H_{k+1}
=
\operatorname{Maestro}\bigl(\Pi,\,H_k\bigr),
}
\]
com
\[
\boxed{
I_k=+1 \Rightarrow \text{expansão }(B\to X),\quad
I_k=0 \Rightarrow \text{retenção }(B\to B),\quad
I_k=-1 \Rightarrow \text{retração }(X\to B).
}
\]
A passagem de volta obedece ao Metrónomo:
\[
\boxed{\lambda^{+} + \lambda^{-} = 0},
\]
de modo que a memória da passagem \textbf{não} constitui perda
estrutural quando a involução fecha. Em suma:
\[
\boxed{
\text{histerese}
=
\text{memória de uma dobra sob orientação do MOVE}.
}
\]
A batuta determina o sentido; o Metrónomo atesta a volta.
\emph{Não há Lei nova:} a histerese é realização composta.
Corpus banal de vizinhança tolerante (erro\,/\,abreviação\,/\,expressão,
sem aula): \code{papers/conversa\_vizinhanca.tex},
\code{tools/vizinhanca.sh}.
\end{teorema}

\begin{proposicao}[núcleo mínimo]\label{prop:histerese-nucleo}
O núcleo operacional do Teorema~\ref{thm:histerese} é
\[
\boxed{
\text{Lei~0 (MOVE)}
\to
\text{Lei~3 (trial }I\text{)}
\to
\text{Lei~4 (dobra / borda)}
\to
\text{Lei~5 (bit / corte)}
\to
\text{Lei~6 (relógio)}.
}
\]
A Lei~1 dá a dualidade da volta; a Lei~2 a orientação rotórica; a
Lei~7 aparece só quando a realização sobe na torre octoniónica.
\end{proposicao}

\begin{tabular}{@{}lp{9.2cm}@{}}
\toprule

\midrule

\end{center}

\begin{corolario}[busca semântica]\label{cor:histerese-busca}
Uma busca com histerese \textbf{não} necessita de dois limiares
independentes como axiomas. A existência de uma borda com direcção
de passagem produz $\mathrm{entrar}\neq\mathrm{sair}$. A tolerância
deixa de ser um número arbitrário e passa a ser \textbf{propriedade
operacional da borda}. A cadeia realizada é
\[
\boxed{
\text{consulta}
\to
\text{massa }X
\to
\text{corte}
\to
\text{suporte}+B
\to
\Pi
\to
\pi_k
\to
\text{Maestro }(I)
\to
\text{Metrónomo }(r).
}
\]
O Maestro não procura a resposta: projecta o suporte. O Metrónomo não
inventa: lê o fechamento. Validação no corpus da assistente:
\code{tools/bench\_histerese.sh} (vizinhança, abreviação, sequência,
borda, $r_{\mathrm{volta}}$).
\end{corolario}

\end{observacao}

\section{Teorema do Controle de Histerese --- dual}\label{sec:controle-histerese}

O Teorema~\ref{thm:histerese} diz \textbf{quando} o estado permanece.
O dual diz \textbf{por onde} se pode procurar uma realização equivalente
sem destruir o estado retido. Continua \textbf{sem Lei~$9$}: o controlo
é operação composta; os eixos são coordenadas de realização, não leis.

\begin{definicao}[vizinhança admissível]\label{def:V-k}
Dado $H_k=(X_k,B_k,I_k)$, a \textbf{vizinhança admissível} $\mathcal{V}_k$
é o produto de eixos de distância (não estruturais):
\[
\boxed{
\mathcal{V}
=
\mathcal{V}_{\mathrm{caixa}}
\times
\mathcal{V}_{\mathrm{teclado}}
\times
\mathcal{V}_{\phi}
\times
\mathcal{V}_{\mathrm{forma}}
\times\cdots
}
\]
com vector de distância
\[
\boxed{
D(x,y)=\bigl(d_{\mathrm{caixa}},\,d_K,\,d_{\phi},\,d_{\mathrm{forma}},\,\ldots\bigr)
}
\]
e região
\[
\boxed{
\mathcal{V}(x)=\{y:D(x,y)\preceq\Theta\}.
}
\]
Eixos (realizações, não leis): \emph{caixa} (segmentação/acentos:
\texttt{cafe}$\leftrightarrow$\texttt{café}); \emph{teclado} (geometria
$d_K$ das teclas); \emph{fonético} $\phi$ (grafemas$\to$fonemas,
$d_{\phi}$).
\end{definicao}

\begin{definicao}[operador de controlo]\label{def:C-k}
\[
\boxed{
\mathcal{C}_k\colon H_k\times\mathcal{V}_k\longrightarrow H_{k+1}\ \text{ou}\ H_k.
}
\]
A histerese fornece a borda $B_k$; o controlo escolhe $v\in\mathcal{V}_k(B_k)$.
\end{definicao}

\begin{teorema}[Controle de Histerese]\label{thm:controle-histerese}
A geração de variantes de um estado ocorre \textbf{dentro} da vizinhança
determinada pela sua borda, segundo eixos de distância admissíveis; a
selecção de uma variante \textbf{não} altera o estado retido enquanto o
critério de fechamento do Metrónomo permanecer satisfeito:
\[
\boxed{
\mathcal{C}_k(v)\in B_k
\;\Rightarrow\;
\text{estado preservado},
}
\qquad
\boxed{
\mathcal{C}_k(v)\notin B_k
\;\Rightarrow\;
\text{retração ou mudança de andar}.
}
\]
Dualidade:
\[
\boxed{
\text{Histerese}=\text{retenção}
\qquad\Longleftrightarrow\qquad
\text{Controle}=\text{seleção}.
}
\]
\[
\boxed{
\text{Histerese: }X\to H;
\qquad
\text{Controle: }H\to X'.
}
\]
Em cadeia:
\[
\boxed{
\operatorname{Gerar}
\to
\operatorname{Medir}(D)
\to
\operatorname{Seleccionar}_{\Pi}
\to
\operatorname{Reter\ /\ Fechar}.
}
\]
A geração filtrada pela partitura evita explosão combinatória:
\[
\boxed{
\operatorname{Var}_{\Pi}(x)
=
\operatorname{Gerar}(x)
\cap
\operatorname{Admissível}(\Pi).
}
\]
\[
\boxed{
\text{Histerese}=\text{memória da dobra};
\qquad
\text{Controle}=\text{movimento admissível pela dobra}.
}
\]
\emph{Prova (esboço).} A borda $B_k$ e $I_k$ vêm do Teorema~\ref{thm:histerese}
(Leis~0,3,4,5,6). $\mathcal{V}_k$ não acrescenta operador estrutural: só
coordena distâncias de realização. $\mathcal{C}_k$ compõe
$\operatorname{Maestro}(\Pi,\cdot)$ com a selecção em $\mathcal{V}_k$; o
Metrónomo atesta o fecho ($\lambda^{+}+\lambda^{-}=0$). Ultrapassar $B_k$
é $I=\pm1$ já previsto --- mudança de andar, não Lei nova.
\end{teorema}

\begin{center}\small
\begin{tabular}{@{}ll@{}}
\toprule
Histerese & Controle \\
\midrule
retém estado & escolhe variante \\
borda $B_k$ & região admissível $\mathcal{V}_k$ \\
memória & geração $\operatorname{Var}$ \\
volta & selecção \\
$\lambda^{+},\lambda^{-}$ & direcção de exploração \\
Metrónomo & critério de fechamento \\
Maestro & execução do controlo \\
\bottomrule
\end{tabular}
\end{center}

\begin{corolario}[torre da histerese]\label{cor:torre-histerese}
Os andares $H_k$ formam uma torre operacional: cada eixo de $\mathcal{V}$
pode refinar-se sem criar dimensão estrutural nova. Hardcore (eixos
fortes) e geração automática de variantes são \textbf{realizações} de
$\mathcal{C}_k$, não postulados. Bandas 8--9 do corpus
(\code{tools/vizinhanca.sh}: ambiguidade controlada; reparação) exercitam
adaptar e soltar sem colar. Leitura polar do mesmo $H_k$:
\S\ref{sec:pares-complexos} (Metrónomo entre cartesiano e polar:
\S\ref{sec:metronomo-ponte}). Realização neural do dual:
\S\ref{sec:rede-dual} --- depois da ponte.

\end{corolario}

\section{Teorema da Absorção --- o estresse pelo endereço}\label{sec:absorcao}

O Metrónomo atesta a volta íntegra: $\lambda^{+}+\lambda^{-}=0$
(Teorema~\ref{thm:metronomo}). Falta a pergunta dual: quanto \textbf{custa}
uma perturbação? Se um passo perturba $k$ elementos de um corpo de $n$
endereços ($\lambda^{+}=k$) e a volta lê o resíduo $R$ ($\lambda^{-}=R$),
o \textbf{estresse} é o excesso
\[
\boxed{e \;=\; R - k.}
\]
A lei exige $e=0$: a leitura não pode dissipar mais do que a perturbação
valeu. E a exigência decide-se pela \textbf{régua de leitura} --- que é o
tema deste teorema, não a robustez do objecto.

\begin{definicao}[duas leituras do mesmo corpo]\label{def:leitura-endereco}
Um corpo de $n$ elementos endereçados admite duas leituras da volta:
\begin{enumerate}
\item \textbf{por posição} --- comparam-se as listas \emph{ordenadas};
a posição de cada elemento na ordem derivada participa da medida;
\item \textbf{pelo endereço} --- cada elemento lê-se pelo seu endereço
próprio (a assinatura que \emph{cavalga o corpo}; \emph{Arquitetura},
teorema da assinatura), e a medida conta faltantes, excedentes e
alterados --- a ordem \textbf{não} participa, porque é derivada do
endereço.
\end{enumerate}
A segunda é a leitura \textbf{dual}: a involução endereço$\to$elemento
fecha sem passar pela lista.
\end{definicao}

\begin{teorema}[Absorção --- o estresse tem teto, e é zero]\label{thm:absorcao}
Na leitura pelo endereço, subindo a torre ($n$ por dobra --- Lei~4):
\begin{enumerate}
\item \textbf{Endereço preservado} (apagar, alterar conteúdo, alterar
categoria, reordenar): $e=0$ \textbf{exato} em todo andar ---
$\lambda^{+}+\lambda^{-}=0$ em contagens inteiras, sem um double.
\item \textbf{Endereço destruído} (o elemento vira ilegível): $e=1$
\textbf{constante} com variância zero --- o elemento destruído conta como
o \textbf{par} faltante$\oplus$excedente, a Lei~0 a nomear os dois nulos
($0=(+1)\oplus(-1)$): quem mata um endereço paga a morte \emph{e} o
nascimento do lixo, e mais nada.
\item \textbf{Teto dimensional}: a variância do estresse \textbf{não
cresce} pela torre em nenhuma classe de perturbação --- a medida escala
sem teto de dimensão a rasgá-la.
\end{enumerate}
Na leitura \emph{por posição}, a classe (2) \textbf{diverge}
$\sim n^{2}$ (média $n/2$): o elemento ilegível perde o lugar na ordem e
empurra tudo entre a sua posição e o fim. A dissipação é da
\textbf{régua}, não do objecto --- a ordem é derivada do endereço, e
\emph{a régua não transporta; a volta transporta}.
\[
\boxed{
\text{endereço}=\text{leitura dual: }e=0;\qquad
\text{posição}=\text{régua derivada: dissipa }\sim n^{2}.
}
\]
\emph{Prova (operacional, medida).} \code{tools/lyapunov\_measure.js}:
inteiro puro, gerador congruente determinístico, \textbf{uma} perturbação
com \textbf{duas} leituras (as réguas medem o mesmo objecto), $T=100$ por
célula, torre $134\to4286$ por dobra sobre o cristal recuperado. Medido:
$e=0$ exato em todas as células de endereço preservado; $e\equiv1$
($\sigma^{2}=0$) no endereço destruído lido pelo endereço; e por posição
$\sigma^{2}\cdot T^{2}$ sobe de $13\,435\,000$ para $16\,130\,133\,019$
quando $n$ sobe $32\times$ --- $n^{2}$ manda $1024\times$, mediu-se
$1200\times$, média $e\approx n/2$. Matriz de perturbação no medidor v2
(\code{tests/cristal\_volta.js} \S V3, as duas réguas na mesma cópia):
permutação e retoque da projeção \emph{absorvidos} ($R=0$); conteúdo,
categoria, remoção e duplicação $R=1$; endereço destruído --- posição
$R=1641$, endereço $R=2$, o par. $\square$
\end{teorema}

\begin{proposicao}[a involução dentro do ruído]\label{prop:involucao-ruido}
Num lote de $m$ perturbações independentes, duas que caem no
\textbf{mesmo endereço} aniquilam-se --- a alteração involutiva compõe
consigo própria e dá a identidade ($\mathrm{flip}\circ\mathrm{flip}=
\mathrm{id}$: \textbf{Lei~1}). O estresse aparece então \emph{negativo},
com probabilidade de colisão $\sim\binom{m}{2}/n$, que \textbf{cai} ao
subir a torre. Não é dissipação: é a involução a fechar dentro do próprio
ruído (medido no mesmo \code{lyapunov\_measure.js}: lote de $8$,
$e<0$ só por colisão, $e\to0$ com $n$).
\end{proposicao}

\begin{corolario}[variante admissível pelo endereço]\label{cor:variante-endereco}
No Controle de Histerese (Teorema~\ref{thm:controle-histerese}), o
critério da vizinhança $\mathcal{V}$ ganha a forma medida:
\[
\boxed{
\text{variante que preserva o endereço}
\;\Rightarrow\;
\mathrm{RETAIN}\ \text{sem custo } (e=0);
}
\]
\[
\boxed{
\text{variante que destrói o endereço}
\;\Rightarrow\;
\mathrm{RETRACT}\ \text{com custo constante (o par), nunca crescente}.
}
\]
O sistema \textbf{distingue} variantes admissíveis das que violam o
invariante --- não vale «toda alteração é corrupção»: reordenar é
variante equivalente ($e=0$ por construção, porque a ordem é derivada), e
o reticulado continua inteiro.
\end{corolario}

\begin{observacao}[estatuto]
Teorema \textbf{operacional}: enuncia o que a medida fechou na classe
considerada (corpos endereçados; perturbações das cinco classes; torre
por dobra até $4286$). Não afirma robustez universal de todo corpo ---
afirma que \emph{o estresse é propriedade da régua de leitura}, e que a
leitura dual (endereço) o absorve. Sem lei nova: Lei~0 (o par dos nulos),
Lei~1 (involução no ruído), Lei~4 (a torre por dobra), Metrónomo (a
atestação $\lambda^{+}+\lambda^{-}=0$).
\end{observacao}

\section{Leitura complexa --- pares, polar, modelo de teste}\label{sec:pares-complexos}

A torre iterada $T_1=i$, $T_{k+1}=i^{T_k}$ é uma
\textbf{analogia dinâmica limpa} --- \emph{não} um teorema do Corpo de
Peano. O Maestro realiza retração entre andares; a torre de $i$ itera
até um ponto fixo. A relação legítima é dinâmica: ambos transformam
estado a cada tick e perguntam se a órbita fecha, converge ou diverge.
Antes de escrever $e^{i\theta}$ ou $i^{z}$, deriva-se Euler
\textbf{neste} paper (\S\ref{sec:euler-peano}), no par conjugado do
andar.

\subsection*{Princípio dos pares complexos}

A ideia \textbf{não} é que cada andar inteiro seja $\mathbb{C}$. É que
cada \textbf{par conjugado} de componentes admite estrutura complexa:
\[
\boxed{(x,y)\;\longleftrightarrow\; z=x+iy}.
\]

\begin{definicao}[par conjugado do andar]\label{def:par-RI}
Para cada $k$, seja
\[
\boxed{P_k = A_k^{(R)}\oplus A_k^{(I)}}.
\]
Se existe unidade $J_k$ com $J_k^{2}=-1$, define-se
\[
\boxed{\Phi_k(x,y)=x+J_k y}.
\]
\end{definicao}

\begin{proposicao}[duas congruências]\label{prop:duas-cong}
Como espaço real bidimensional,
\[
\boxed{P_k\cong_{\mathbb{R}}\mathbb{C}}
\]
é automático. Já
\[
\boxed{P_k\cong_{\mathrm{alg}}\mathbb{C}}
\]
exige que a multiplicação de $A_k$ seja compatível com $J_k$
($i^{2}=-1$ e a regra $(x_1+iy_1)(x_2+iy_2)=\cdots$). Identificação
geométrica $\neq$ isomorfismo algébrico --- a auditoria mantém a
distinção explícita.
\end{proposicao}

A diferença entre andares não é a \emph{espécie} do par, mas a
\textbf{escala/resolução} da realização. A retração $\pi_k\colon P_{k+1}\to P_k$
é compatível com a realização complexa quando existe $\rho_k$ com
\[
\boxed{\Phi_k\circ\pi_k \;=\; \rho_k\circ\Phi_{k+1}}.
\]
Ou seja: a dobra \textbf{não} destrói o par; projecta-o para a escala
seguinte. Em diagrama:
\[
\boxed{
\begin{array}{ccccc}
P_0 & \leftarrow & P_1 & \leftarrow & P_2\leftarrow\cdots \\
\downarrow\Phi_0 && \downarrow\Phi_1 && \downarrow\Phi_2 \\
\mathbb{C} & \xleftarrow{\rho_0} & \mathbb{C}
& \xleftarrow{\rho_1} & \mathbb{C}\leftarrow\cdots
\end{array}
}
\]
\[
\boxed{
\text{andar}\to\text{par }(R,I)\to\mathbb{C};
\qquad
\text{torre}=\text{retrações entre essas representações}.
}
\]
O próximo passo mensurável: construir a matriz de cada $\pi_k$ no par
$(R,I)$ e verificar se comuta com a multiplicação por $i$ ---
\emph{testar}, não apenas afirmar, a compatibilidade.

\subsection*{Derivação de Euler no par}\label{sec:euler-peano}

No par $P_k$ a unidade $J=J_k$ satisfaz $J^{2}=-1$
(Def.~\ref{def:par-RI}). O fluxo gerado por $J$ é a série
\[
\boxed{
\exp(tJ)
\;=\;
\sum_{n\geq0}\frac{(tJ)^{n}}{n!}.
}
\]
Como $J^{2}=-1$, as potências ciclam com período quatro:
\[
\boxed{
J^{0}=1,\quad
J^{1}=J,\quad
J^{2}=-1,\quad
J^{3}=-J,
\quad\text{e depois repetem}.
}
\]
Separando a série pelos índices pares e ímpares:
\[
\exp(tJ)
\;=\;
\sum_{m\geq0}\frac{(-1)^{m}t^{2m}}{(2m)!}\,1
\;+\;
\sum_{m\geq0}\frac{(-1)^{m}t^{2m+1}}{(2m+1)!}\,J.
\]
Define-se
\[
\boxed{
c(t)=\sum_{m}\frac{(-1)^{m}t^{2m}}{(2m)!},
\qquad
s(t)=\sum_{m}\frac{(-1)^{m}t^{2m+1}}{(2m+1)!},
}
\]
donde
\[
\boxed{\exp(tJ) \;=\; c(t)\,1 \;+\; s(t)\,J}.
\]

\begin{proposicao}[conservação da cruz]\label{prop:euler-cruz}
Derivando termo a termo, $c'=-s$ e $s'=c$, com $c(0)=1$, $s(0)=0$.
A Lei~$2$ (rotor / não-dissipação) dá
\[
\bigl(c^{2}+s^{2}\bigr)'
\;=\;
2cc'+2ss'
\;=\;
-2cs+2sc
\;=\;
0,
\]
logo
\[
\boxed{c(t)^{2}+s(t)^{2}\;\equiv\;1}.
\]
Escreve-se $\cos t:=c(t)$ e $\sin t:=s(t)$ --- \textbf{não} por
postular o círculo: o círculo é o rasto $\{c(t)+s(t)J\}$ no plano do
par. Unicidade: o par $(c,s)$ resolve o IVP linear no plano $(R,I)$.
\end{proposicao}

\begin{teorema}[Euler no Corpo de Peano]\label{thm:euler-peano}
Na realização $J\mapsto i$ do par conjugado
(Prop.~\ref{prop:duas-cong}),
\[
\boxed{e^{i\theta} \;=\; \cos\theta \;+\; i\sin\theta}.
\]
Em particular
\[
\boxed{i^{z} \;=\; e^{\log(i)\cdot z} \;=\; e^{i\pi z/2}},
\]
porque $i=e^{i\pi/2}$ segue de $\cos(\pi/2)+i\sin(\pi/2)=i$.
Assim $F(z)=i^{z}=e^{i\pi z/2}$ \textbf{deriva-se} neste paper; não
se importa como metáfora.
\end{teorema}

\emph{Prova.} O Teorema reduz-se à Proposição~\ref{prop:euler-cruz}
com $J=i$: $\exp(i\theta)=c(\theta)+i\,s(\theta)$. Os valores
$\cos(\pi/2)=0$, $\sin(\pi/2)=1$ dão $e^{i\pi/2}=i$; a identidade
$a^{b}=e^{b\log a}$ no ramo principal de $\mathbb{C}^{\times}$ dá
$i^{z}=e^{z\cdot i\pi/2}$. $\square$

\noindent Corolários na torre de teste:
\[
\boxed{e^{i\pi}=-1}
\quad\text{(Lei~$1$ no fluxo)},
\qquad
\boxed{z=r e^{i\theta}}
\quad\text{(forma polar do mesmo par)}.
\]
A ponte à fundação: a mesma decomposição aparece em
\code{teoria.tex} na definição dinâmica de $\pi$; aqui ela é
\textbf{leitura do andar} --- o que o par $(R,I)$ precisa para o
modelo $T_{k+1}=i^{T_k}$ ser legítimo.

\subsection*{Maestro $\neq$ operador $F$}

Seja $F(z)=e^{i\pi z/2}=i^{z}$ (Teorema~\ref{thm:euler-peano}).
Então $T_k=F^{k-1}(i)$ e o limite
$T_{\ast}$ satisfaz $F(T_{\ast})=T_{\ast}$. Correspondência conceitual:
\begin{center}\small
\begin{tabular}{@{}ll@{}}
\toprule
torre de $i$ & dinâmica do Maestro \\
\midrule
$T_k$ & estado no tick $k$ \\
$F(z)=i^{z}$ & operador de evolução (modelo) \\
$k\mapsto k+1$ & tick \\
$F(T)=T$ & estado que fecha \\
convergência & estabilização da órbita \\
\bottomrule
\end{tabular}
\end{center}
Mas
\[
\boxed{\text{Maestro: retração }\pi_k\colon A_{k+1}\to A_k}
\qquad\text{enquanto}\qquad
\boxed{\text{torre de }i\text{: iteração }F\colon\mathbb{C}\to\mathbb{C}}.
\]
O Metrónomo \textbf{não} é $F$: conta e atesta iterações
($k\mapsto k+1$; medição). O Maestro, no modelo de teste, realiza
$T_k\mapsto T_{k+1}=F(T_k)$. O Maestro do paper
(Teorema~\ref{thm:proj-maestro}) continua
$\mathcal{P}_k=\mathrm{tick}\circ\mathrm{batuta}\circ\Pi$ --- projecta
$\pi_k$; \textbf{não} inventa o próximo estado.

\begin{proposicao}[separação]\label{prop:analogia-i}
A dinâmica $T_{k+1}=i^{T_k}$ \textbf{não prova} a torre dos andares.
É \textbf{modelo de teste}: pergunta, andar a andar, se a estrutura
postulada satisfaz simultaneamente par complexo, retração, histerese e
fecho. Sem essa separação, a analogia vira teorema falso.
\end{proposicao}

\subsection*{Dinâmica cartesiana do modelo}

Sobre $z_k=x_k+iy_k$, a operação $z_{k+1}=i^{z_k}=e^{i\pi z_k/2}$
(Teorema~\ref{thm:euler-peano}) separa os eixos:
\[
\boxed{
x_{k+1}=e^{-\pi y_k/2}\cos\frac{\pi x_k}{2},
\qquad
y_{k+1}=e^{-\pi y_k/2}\sin\frac{\pi x_k}{2}.
}
\]
\textbf{Real = um eixo; imaginário = o outro} --- o tipo de par que o
Maestro pode projectar.

\subsection*{Metrónomo --- ponte entre cartesiano e polar}\label{sec:metronomo-ponte}

O Teorema~\ref{thm:metronomo} \textbf{não} escolhe coordenadas: lê a
volta. Entre as duas visões do mesmo estado $z=x+iy=re^{i\theta}$, o
Metrónomo é a ponte:
\[
\boxed{
\begin{array}{ccc}
\text{visão cartesiana }(x,y)
&\;\xrightarrow{\;\text{Metrónomo}\;}\;
&
\text{visão polar }(r,\theta)
\\
\Delta x,\,\Delta y
&\;\longleftrightarrow\;
&
\Delta r,\,\Delta\theta
\\
\text{frente/volta }(\lambda^{+},\lambda^{-})
&\;\text{atesta}\;
&
\text{fecho da órbita}
\end{array}
}
\]
\[
\boxed{
\text{Maestro projecta o estado};
\qquad
\text{Metrónomo mede o tick nas duas leituras};
\qquad
\text{não inventa estrutura}.
}
\]
No cartesiano, o tick lê o deslocamento do par $(R,I)$; no polar, o
mesmo tick decompõe-se em
\[
\boxed{\text{Metrónomo}\longrightarrow(\Delta r_k,\,\Delta\theta_k)}.
\]
A condição de fecho é a mesma lei ($\lambda^{+}+\lambda^{-}=0$), lida
em qualquer das duas cartas:
\[
\boxed{
\Delta x\to0,\;\Delta y\to0
\quad\Longleftrightarrow\quad
\Delta r\to0,\;\Delta\theta\to0\pmod{2\pi}
}
\]
quando a mudança de coordenadas é regular fora da origem. Assim o
Metrónomo \textbf{fica entre} as duas visões: traduz a medição, não a
dinâmica. A dinâmica (Maestro / $F$) muda de carta; a atestação
permanece.

\subsection*{Forma polar --- módulo e fase}

Escreva $T_k=r_k e^{i\theta_k}$ (corolário de Euler). Então
\[
\boxed{
r_{k+1}=e^{-\frac{\pi r_k}{2}\sin\theta_k},
\qquad
\theta_{k+1}=\frac{\pi r_k}{2}\cos\theta_k\pmod{2\pi}.
}
\]
O próximo módulo depende do seno da fase; a próxima fase, do cosseno.
Em linguagem da torre:
\[
\boxed{r=\text{amplitude/ocupação},
\qquad
\theta=\text{fase/orientação}.}
\]
Isto é mais próximo da histerese do que o cartesiano puro.

\begin{proposicao}[dualidade cruzada]\label{prop:cruzada}
Como $r_{k+1}=e^{-\pi y_k/2}$ e $\theta_{k+1}=\pi x_k/2$,
\[
\boxed{y_k\to r_{k+1},
\qquad
x_k\to\theta_{k+1}.}
\]
O eixo imaginário controla o módulo seguinte; o real, a fase seguinte.
O par $(R,I)$ \textbf{não} desaparece no polar: muda de função,
\[
\boxed{(R,I)\;\longrightarrow\;(\theta,r)}
\quad\text{com}\quad
R\to\theta_{\mathrm{próximo}},\;
I\to r_{\mathrm{próximo}}.
\]
\end{proposicao}

\begin{definicao}[histerese polar]\label{def:H-polar}
Além de $H_k=(X_k,B_k,I_k)$ (Def.~\ref{def:H-k}), a leitura polar
do mesmo estado é
\[
\boxed{
H_k^{\mathrm{pol}}
=
\bigl(r_k,\,\theta_k;\,\Delta r_k,\,\Delta\theta_k;\,I_k\bigr),
}
\]
com região de retenção em anel\,/\,setor
\[
\boxed{
B_k^{\mathrm{pol}}
=
[r_k-\Delta r_k,\,r_k+\Delta r_k]
\times
[\theta_k-\Delta\theta_k,\,\theta_k+\Delta\theta_k].
}
\]
\[
\boxed{\text{histerese polar}=\text{anel}+\text{setor}+\text{memória de sentido}}.
\]
\end{definicao}

\subsection*{Ponto fixo do modelo (após a ponte)}

Com o Metrónomo a ler $(\Delta r,\Delta\theta)$
(\S\ref{sec:metronomo-ponte}), o fecho não é ``só o raio parou'':
\[
\boxed{\Delta r_k\to0
\quad\text{e}\quad
\Delta\theta_k\to0\pmod{2\pi}}.
\]
No ponto fixo $T_{\ast}=r_{\ast}e^{i\theta_{\ast}}$ de $T=i^{T}$,
\[
\boxed{
r_{\ast}=e^{-\frac{\pi r_{\ast}}{2}\sin\theta_{\ast}},
\qquad
\theta_{\ast}=\frac{\pi r_{\ast}}{2}\cos\theta_{\ast}\pmod{2\pi}.
}
\]
Com $0<\theta_{\ast}<\pi/2$ tem-se $\sin\theta_{\ast}>0$ e
$r_{\ast}<1$ --- \textbf{contração radial natural} nessa região.
Numericamente (modelo de teste, não axioma do paper):
\[
\boxed{
|T_{\ast}|=r_{\ast}\approx 0{,}567559\ldots,
\qquad
\theta_{\ast}\approx 0{,}688462\ldots\ \mathrm{rad},
}
\]
\[
T_{\ast}\approx 0{,}438283+0{,}360599\,i.
\]
A norma \textbf{não} converge para $0$ nem para $1$: entra num
\textbf{atrator interior local} em torno de $r_{\ast}$.
\[
\boxed{
\text{contração}\;\Rightarrow\;\text{retenção local em torno de }T_{\ast}
}
\qquad\text{mas \emph{não}}
\qquad
\boxed{
\text{contração}\;\Rightarrow\;\text{preenchimento do andar}.
}
\]
Preencher o andar exige a banda $B_k^{\mathrm{pol}}$
(Def.~\ref{def:H-polar}) e medir cobertura\,/\,recorrência sob a
dinâmica --- histerese$+$cobertura, não só o ponto fixo.

\subsection*{Releitura do Lyapunov dualizado}\label{sec:lyap-dual-releitura}

Três leituras distintas --- não três nomes para o mesmo número.

\begin{center}\small
\begin{tabular}{@{}llp{6.2cm}@{}}
\toprule
leitura & objecto & o que mede \\
\midrule
clássica & $\lambda=\log|F'|$ & contração local num tick \\
paper (Metrónomo) & $(\lambda^{+},\lambda^{-})$ & frente\,/\,volta (reversão) \\
polar (modelo) & espectro de $J_{\mathrm{polar}}$ & canais $(r,\theta)$ conjugados \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{1. Lyapunov clássico do modelo.}
Como $F'(z)=\frac{i\pi}{2}F(z)$ (Teorema~\ref{thm:euler-peano}),
ao longo da órbita $|F'(T_k)|=\frac{\pi}{2}r_{k+1}$. No limite
$r_k\to r_{\ast}$,
\[
\boxed{
\lambda_{\mathrm{cl}}
\;=\;
\log\Bigl(\frac{\pi}{2}r_{\ast}\Bigr)
\;\approx\;
-0{,}1148
\;<\;0
}.
\]
Contração local em $T_{\ast}$ --- escalar; ainda não é o dualizado do
paper.

\paragraph{2. Lyapunov dualizado do paper --- frente e volta.}
No Teorema~\ref{thm:metronomo} o par é
\[
\boxed{\lambda^{+} \;=\; \log|f'|,\qquad
\lambda^{-} \;=\; \log|(f^{-1})'| \;=\; -\lambda^{+}},
\]
donde
\[
\boxed{\lambda^{+} + \lambda^{-} \;=\; 0}
\]
numa volta fechada (\code{tests/lyapunov\_refletido.c}: metade
refletida). Espectro $\pm$ da classe reversível --- dualidade
\textbf{temporal}, não espectral\,/\,polar.
\[
\boxed{
\text{clássico }=\text{ um canal};
\qquad
\text{dualizado do paper }=\text{ frente}+\text{reflexão}.
}
\]

\paragraph{3. Jacobiano polar concreto.}
Com $a=\pi/2$,
\[
\boxed{
f_r(r,\theta)=e^{-a r\sin\theta},
\qquad
f_{\theta}(r,\theta)=a r\cos\theta.
}
\]
O Jacobiano é
\[
J_{\mathrm{pol}}
=
\begin{pmatrix}
-a\sin\theta\,f_r & -a r\cos\theta\,f_r \\
a\cos\theta & -a r\sin\theta
\end{pmatrix}.
\]
No ponto fixo $f_r=r_{\ast}$,
\[
\boxed{
J_{\mathrm{pol}}(T_{\ast})
=
\begin{pmatrix}
-a r_{\ast}\sin\theta_{\ast} & -a r_{\ast}^{2}\cos\theta_{\ast} \\
a\cos\theta_{\ast} & -a r_{\ast}\sin\theta_{\ast}
\end{pmatrix}.
}
\]
Daí
\[
\boxed{\det J_{\mathrm{pol}} \;=\; a^{2}r_{\ast}^{2} \;=\; |F'(T_{\ast})|^{2}},
\]
correcto. \textbf{Mas} $\det=|F'|^{2}$ \emph{não} implica, por si,
autovalores conjugados: isso vem da forma concreta desta matriz.

\begin{proposicao}[espectro polar em $T_{\ast}$]\label{prop:J-polar}
A matriz $J_{\mathrm{pol}}(T_{\ast})$ é do tipo
$\bigl(\begin{smallmatrix}A&B\\C&A\end{smallmatrix}\bigr)$ com
$A=-a r_{\ast}\sin\theta_{\ast}$ e
$BC=-a^{2}r_{\ast}^{2}\cos^{2}\theta_{\ast}<0$ (em
$0<\theta_{\ast}<\pi/2$). Logo
\[
\boxed{
\mu_{\pm}
\;=\;
A\pm\sqrt{BC}
\;=\;
-a r_{\ast}\sin\theta_{\ast}
\pm i\, a r_{\ast}\cos\theta_{\ast}
}
\]
\[
\boxed{
\mu_{\pm}
\;=\;
a r_{\ast}\bigl(-\sin\theta_{\ast}\pm i\cos\theta_{\ast}\bigr)
\;=\;
\pm i\,|F'(T_{\ast})|\,e^{\pm i\theta_{\ast}}.
}
\]
Em particular
\[
\boxed{|\mu_{\pm}| \;=\; a r_{\ast} \;=\; |F'(T_{\ast})|},
\qquad
\boxed{
\log|\mu_{+}|=\log|\mu_{-}|=\lambda_{\mathrm{cl}}\approx-0{,}1148
},
\]
e $\log|\mu_{+}|+\log|\mu_{-}|=2\lambda_{\mathrm{cl}}$. Numericamente
$\mu_{\pm}\approx-0{,}566\pm0{,}688\,i$, $|\mu|\approx0{,}8915$.
O ``dual'' polar \textbf{não} é um par real distinto: são conjugados
\textbf{porque} diagonais iguais e $BC<0$ neste Jacobiano --- não por
um princípio genérico ``toda realização polar de diferencial complexa''.
\end{proposicao}

\emph{Prova.} Forma de $J_{\mathrm{pol}}$ por derivação de
$(f_r,f_{\theta})$; no ponto fixo $f_r=r$. Autovalores de
$\bigl(\begin{smallmatrix}A&B\\C&A\end{smallmatrix}\bigr)$:
$A\pm\sqrt{BC}$. Com $BC<0$, $\sqrt{BC}=i\,a r_{\ast}|\cos\theta_{\ast}|$;
em $0<\theta_{\ast}<\pi/2$ obtém-se a fórmula acima. Módulo:
$a r_{\ast}\sqrt{\sin^{2}+\cos^{2}}=a r_{\ast}=|F'|$. A identidade
$-\sin\theta\pm i\cos\theta=\pm i\,e^{\pm i\theta}$ dá a forma polar do
espectro. $\square$

\noindent Leitura:
\[
\boxed{
\text{módulo do espectro}\;\longleftrightarrow\;\text{contração},
\qquad
\text{argumento do espectro}\;\longleftrightarrow\;\text{orientação/fase}.
}
\]
A polarização separa ocupação e orientação; a dinâmica mantém-nas
acopladas.

\paragraph{4. Dualidade temporal $\neq$ dualidade espectral.}
\[
\boxed{
(\lambda^{+},\lambda^{-})=(\lambda_{\mathrm{passo}},-\lambda_{\mathrm{passo}}),
\qquad
\lambda^{+}+\lambda^{-}=0
\quad\text{(reversão)}
}
\]
\[
\boxed{
\bigl(\log|\mu_{+}|,\log|\mu_{-}|\bigr)=(\lambda_{\mathrm{cl}},\lambda_{\mathrm{cl}}),
\qquad
\log|\mu_{+}|+\log|\mu_{-}|=2\lambda_{\mathrm{cl}}
\quad\text{(volume infinitesimal)}
}
\]
\[
\boxed{
+\lambda\;\leftrightarrow\;-\lambda
\quad\text{(temporal)}
\qquad\neq\qquad
\mu_{+}\;\leftrightarrow\;\overline{\mu_{-}},\ |\mu_{+}|=|\mu_{-}|
\quad\text{(espectral/polar)}.
}
\]
O modelo $i^{T_k}$ ilustra a dualização polar; a atestação
$\lambda^{+}+\lambda^{-}=0$ continua a ser a do Teorema~\ref{thm:metronomo},
medida em \code{tests/lyapunov\_refletido.c}.

\section{Rede neural dual --- realização do Controle de Histerese}\label{sec:rede-dual}

O Teorema~\ref{thm:controle-histerese} diz \emph{retenção}$\leftrightarrow$\emph{selecção}.
Esta secção realiza esse dual por um sistema híbrido $(P,\mathcal{H},H)$
e eleva o que é demonstrável a teorema --- sem colar Hopfield com
inversão. Vem \textbf{depois} da leitura complexa: o Metrónomo já
ponteia cartesiano e polar (\S\ref{sec:metronomo-ponte}); a rede
realiza o controlo sobre essa mesma atestação. Núcleo: leis já no
corpo (Teorema~\ref{thm:histerese}, Prop.~\ref{prop:histerese-nucleo});
estaca do Dual Sort (\code{papers/dualsort.tex}); Metrónomo
(Teorema~\ref{thm:metronomo}, \S\ref{sec:metronomo-ponte}).

\subsection*{Três níveis (não misturar)}

\begin{center}\small
\begin{tabular}{@{}llp{5.8cm}@{}}
\toprule
nível & objecto & estatuto \\
\midrule
geometria da estaca & $W_P=-I$, $b_P=c$ & determinístico (Lei~1) \\
histerese como controlador & retenção na banda & teorema (indução) \\
memória associativa + volta & $H$, conjugação $D$ & teorema \emph{sob} dualidade reversível \\
\bottomrule
\end{tabular}
\end{center}
Três memórias distintas:
\[
\boxed{
\begin{array}{rcl}
\text{estaca} &\longrightarrow& \text{memória geométrica}\\
\text{histerese} &\longrightarrow& \text{memória de estado}\\
\text{Hopfield} &\longrightarrow& \text{memória associativa}
\end{array}
}
\]
\[
\boxed{
\text{geometria}\to\text{borda}\to\text{retenção}\to\text{atração}.
}
\]

\subsection*{Sistema híbrido}

\begin{definicao}[sistema neural de histerese]\label{def:sistema-hibrido}
O estado verdadeiro é o par
\[
\boxed{X_k=(x_k,h_k)},
\]
não $x_k$ sozinho --- a histerese só existe porque o estado inclui a
memória (Teorema~\ref{thm:histerese}: $H_k=(X_k,B_k,I_k)$).
O controlador perceptron (estaca, Lei~1) é
\[
\boxed{u_k=W_Px_k+b_P},\qquad
\boxed{W_P=-I,\quad b_P=c}
\]
(normalização $u=c-x$; equivalente $W_P=-2I$, $b_P=2c$ pela
$x^{\dagger}-x=2(c-x)$). A memória de estado actualiza-se por
\[
\boxed{
h_{k+1}
=
\begin{cases}
+1 & u_k>\Delta,\\
h_k & |u_k|\le\Delta,\\
-1 & u_k<-\Delta.
\end{cases}
}
\]
A aplicação do sistema é
\[
\boxed{\mathcal{F}\colon(x,h)\mapsto(x',h')},
\]
com $u=c-x$ e
\[
\boxed{
\begin{cases}
|u|\le\Delta & \Longrightarrow\; (x',h')=(x,h)
\quad\text{(retenção)},\\[4pt]
u>\Delta & \Longrightarrow\;
x'=F_P(x),\; h'=+1
\quad\text{(frente)},\\[4pt]
u<-\Delta & \Longrightarrow\;
x'=F_H(x),\; h'=-1
\quad\text{(volta dual)},
\end{cases}
}
\]
onde $F_H=D\circ F_P^{-1}\circ D^{-1}$
(Def.~\ref{def:conjugacao-D}) e $D(x)=2c-x$ (estaca).
Hopfield fornece a memória associativa $Y$, \textbf{não} define $x'$.
A banda de ambiguidade é
\[
\boxed{\mathcal{B}_{\Delta}=[c-\Delta,\,c+\Delta]},
\qquad
\boxed{b_{-}-b_{+}=2\Delta}
\]
(dois limiares duais com $W_{\pm}=-I$; Prop.~\ref{prop:dois-limiares}).
\end{definicao}

\begin{proposicao}[pesos pela Lei~1]\label{prop:pesos-lei1}
A estaca $x^{\dagger}=2c-x$ (involução, Lei~1; Dual Sort) determina
\[
\boxed{W_P=-I=\text{geometria}},
\qquad
\boxed{b_{-}-b_{+}=2\Delta=\text{memória de retenção}}.
\]
O peso \textbf{não} precisa ser aprendido para existir a borda. A
aprendizagem, se houver, fica em $W_{H}$ (memória associativa), não em
$W_P$. Fundamento: Lei~1 (dual/involução); Lei~3 (trial
$I\in\{-1,0,+1\}$ na banda); Lei~4 (dobra/borda,
Def.~\ref{def:borda}); Lei~0 (MOVE sem comparação destrutiva).
\end{proposicao}

\begin{proposicao}[dois limiares duais]\label{prop:dois-limiares}
\[
\boxed{u_{+}(x)=c-\Delta-x,\qquad u_{-}(x)=c+\Delta-x},
\]
\[
\boxed{w_{+}=w_{-}=-1,\qquad b_{+}=c-\Delta,\qquad b_{-}=c+\Delta}.
\]
Na região $c-\Delta<x<c+\Delta$: $P_{+}(x)\neq P_{-}(x)$ ---
\[
\boxed{\text{discordância dos duais}=\text{região de histerese}}.
\]
\end{proposicao}

\begin{proposicao}[unidade ternária com memória]\label{prop:trial-neural}
O trial $\{-1,0,+1\}$ (Lei~3) realiza-se por
\[
\boxed{
s(x)=\operatorname{sgn}(c-x),\qquad
s=0\;\Longleftrightarrow\; x^{\dagger}=x.
}
\]
Com histerese no trial: $s_{k+1}=s_k$ quando $|x_k-c|\le\Delta$ --- a
memória é o estado que a borda impede de mudar
(Teorema~\ref{thm:histerese}).
\end{proposicao}

\begin{teorema}[Retenção Neural]\label{thm:retencao-neural}
Uma rede limiar com dois limiares separados por $2\Delta$
(Def.~\ref{def:sistema-hibrido}) conserva o estado interno sob qualquer
sequência de perturbações confinada à banda de histerese:
\[
\boxed{|u_k|\le\Delta\;\Longrightarrow\; h_{k+1}=h_k}.
\]
Em particular, se $u_0,\ldots,u_{n-1}\in[-\Delta,\Delta]$, então
\[
\boxed{h_n=h_0}.
\]
\end{teorema}

\emph{Prova.} Por definição de $h_{k+1}$, $|u_k|\le\Delta$ força
$h_{k+1}=h_k$. Indução em $k$ dá $h_n=h_0$. $\square$

\noindent Fundamento: Teorema~\ref{thm:histerese}
($I=0\Rightarrow$ retenção $B\to B$); Teorema~\ref{thm:controle-histerese}

\subsection*{Memória associativa $\neq$ inversão}

\begin{definicao}[três operadores]\label{def:tres-ops}
\[
\boxed{
\underbrace{P}_{\text{percepção}}
\;\longleftrightarrow\;
\underbrace{\mathcal{H}}_{\text{borda}}
\;\longleftrightarrow\;
\underbrace{H}_{\text{memória}}
}
\]
são \textbf{três} operadores distintos. \textbf{Proibição:}
\[
\boxed{
\text{Hopfield realiza a memória da volta;}
\quad
\text{a dualidade reversível é que garante }\lambda^{-}.
}
\]
Hopfield clássico ($W_H=\frac1N\sum_{\mu}\xi^{\mu}(\xi^{\mu})^{\!\top}$,
$(W_H)_{ii}=0$) dá \emph{retorno à memória} (atração), não
automaticamente \emph{retorno inverso da trajectória}. Atracção $\neq$
reversibilidade.
\end{definicao}

\begin{definicao}[conjugação reversível]\label{def:conjugacao-D}
Seja $F_P$ a dinâmica de decisão (frente) e $F_H$ uma realização da
volta. Diz-se que $F_H$ é \textbf{dual conjugado} a $F_P$ no conjunto
admissível $\mathcal{A}$ se existe $D$ com
\[
\boxed{
F_H \;=\; D\circ F_P^{-1}\circ D^{-1}
\quad\text{em }\mathcal{A}.
}
\]
Então
\[
\boxed{DF_H \;=\; D\,(DF_P)^{-1}\,D^{-1}},
\]
e os invariantes locais passam aos inversos.
\end{definicao}

\begin{teorema}[Realização Neural Dual da Histerese]\label{thm:rede-dual}
Considere o sistema híbrido $X_k=(x_k,h_k)$
(Def.~\ref{def:sistema-hibrido}) com controlador $u_k=-x_k+c$ e banda
$\mathcal{B}_{\Delta}$. Suponha que exista realização dual $F_H$
conjugada a $F_P$ no admissível $\mathcal{A}$
(Def.~\ref{def:conjugacao-D}), com métrica compatível com $D$. Então:
\begin{enumerate}
\item \textbf{Retenção.} $|u_k|\le\Delta\Rightarrow h_{k+1}=h_k$
(Teorema~\ref{thm:retencao-neural}).
\item \textbf{Borda.} A região de ambiguidade é
$\mathcal{B}_{\Delta}=[c-\Delta,c+\Delta]$
(Def.~\ref{def:borda}; Prop.~\ref{prop:dois-limiares}).
\item \textbf{Dualidade dinâmica.} $DF_H=(DF_P)^{-1}$ até conjugação
por $D$.
\item \textbf{Lyapunov dualizado.}
\[
\boxed{\lambda_H=-\lambda_P},
\qquad
\boxed{\lambda_P+\lambda_H=0},
\]
com
\[
\lambda_P=\log|DF_P|,\qquad
\lambda_H=\log|DF_H|
\]
(leitura do Teorema~\ref{thm:metronomo} no par frente/volta).
\item \textbf{Fecho.} Se $h_n=h_0$ e a composição fecha em $\mathcal{A}$,
\[
\boxed{F_H\circ F_P=\mathrm{id}_{\mathcal{A}}}.
\]
\end{enumerate}
Em cadeia:
\[
\boxed{
P
\overset{\lambda^{+}}{\longrightarrow}
\text{borda/histerese}
\overset{\lambda^{-}}{\longrightarrow}
\text{memória},
\qquad
\lambda^{+}+\lambda^{-}=0.
}
\]
\emph{Não} há Lei~$9$: tudo compõe Leis~0--6, Teoremas~\ref{thm:histerese},
\ref{thm:controle-histerese}, \ref{thm:metronomo},
\ref{thm:proj-maestro} (Maestro escolhe $F_P$ ou $F_H$ via batuta
$I$; Metrónomo atesta o fecho).
\end{teorema}

\emph{Prova.} (i)~Teorema~\ref{thm:retencao-neural}. (ii)~Definição da
banda e Prop.~\ref{prop:dois-limiares} (Lei~1$+$Lei~4). (iii)~Derivada
da conjugação $F_H=D\circ F_P^{-1}\circ D^{-1}$. (iv)~Tomando módulos
(ou log-norma) compatíveis com $D$:
$\log|DF_H|=-\log|DF_P|$. Isto é o espectro $\pm$ do
Teorema~\ref{thm:metronomo} realizado no par $(F_P,F_H)$, não um
número inventado. (v)~Em $\mathcal{A}$,
$F_H\circ F_P=D\circ F_P^{-1}\circ D^{-1}\circ F_P$; com $D$ a
identificar a leitura da volta (involução Lei~1 / estaca), a composição
fecha. O Maestro (Teorema~\ref{thm:proj-maestro}) projecta a escolha
de lado; o controlo (Teorema~\ref{thm:controle-histerese}) restringe a
variante a $\mathcal{V}_k$ / $\mathcal{B}_{\Delta}$. $\square$

\noindent \textbf{Rotor (Lei~2).} No par,
$W_{\mathrm{estaca}}=-I$ (reflexão) e
$W_{\mathrm{rotor}}=J$ com $J^{2}=-I$ (rotação): duas faces da mesma
dualidade (Lei~1$\leftrightarrow$Lei~2).

\begin{proposicao}[fecho experimental da conjugação]\label{prop:fecho-conj-exp}
No modelo afim $F_P(x)=c+A(x-c)$ com $A$ invertível e
$D(x)=2c-x$, a realização
$F_H=D\circ F_P^{-1}\circ D^{-1}$ verifica numericamente
\[
\boxed{DF_H\,DF_P=I},
\qquad
\boxed{F_H\circ F_P=\mathrm{id}},
\qquad
\boxed{\lambda_P+\lambda_H=0}
\]
em dimensão~$1$ e~$2$ (incluindo shear), por diferenças finitas
(\code{tests/rede\_dual.js} §R5--§R7). O sistema híbrido
$\mathcal{F}\colon(x,h)\mapsto(x',h')$ conserva $h$ sob perturbações
na banda (§R6; Teorema~\ref{thm:retencao-neural}).
Hopfield fornece $Y$ (memória), \textbf{não} substitui $F_H$.
\end{proposicao}

\noindent Isto fecha os quatro pontos do gerente:
(1)~$\mathcal{F}$ híbrido; (2)~retenção; (3)~Hopfield~$\neq$~inversão;
(4)~conjugação reversível medida. A hipótese algébrica
(Def.~\ref{def:conjugacao-D}) permanece; o experimento atesta a
realização no modelo Dual Sort / estaca.

\noindent \textbf{Ponte ao Teorema Central.} Ver
Prop.~\ref{prop:central-operacional}: quatro camadas
($\pi_k$, Maestro/Metrónomo, controlo, música).
A rede dual \textbf{não} é o Teorema Central; a batuta
\textbf{não} é $I$. Em cadeia:
\[
\boxed{
\text{Teorema Central}
\;\longrightarrow\;
\text{Maestro/Metrónomo}
\;\longrightarrow\;
\text{controlo }(P,\mathcal{H},H)
\;\longrightarrow\;
\mathrm{Conservat\acute{o}rio}\!\to\!\mathrm{Pera}\!\to\!\mathrm{Batuta}.
}
\]
Prop.~\ref{prop:rede-no-canal}: $I$ da batuta alimenta $\mathcal{F}$.
