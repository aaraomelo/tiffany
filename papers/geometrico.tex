\begin{abstract}
\noindent Este paper é \textbf{filho} do \emph{Corpo Universal}. Não
reconstrói o sistema nem decide se ele é permitido: \textbf{realiza
geometricamente a construção que o Universal já fechou}, na ordem
\[
\boxed{\;
\text{8 leis}\to\text{base ortonormal}\to\text{bit a bit}\to\text{dual}
\to\text{recorrência}\to\text{Pisot}\to\text{encaixe}\to\text{corte}
\to\mathbb{R}\to\pi_k .\;}
\]
Cada seta deixa um objecto no papel e um medidor no repositório. A base das
oito leis tem Gram $=I$, e por isso $b_i=\langle b,e_i\rangle$ \emph{é} o bit
--- o que autoriza o bit a bit não é a codificação, é a ortonormalidade. O dual
$\sigma\dg=-\sigma^{-1}$ produz a recorrência inteira; o operador \textbf{produz}
o inteiro $t_k=\sigma^{k}+(\sigma\dg)^{k}$, e com ele
$\operatorname{dist}(\sigma^{k},\mathbb{Z})=\abs{\sigma\dg}^{k}$: a aproximação
não é escolhida. Daí o encaixe, e daí o corte.

\smallskip\noindent
E a construção é de $\mathbb{R}$ \textbf{inteiro}. Corre sobre quocientes
parciais \emph{arbitrários} --- um real \emph{é} uma dessas sucessões ---, e
o metálico é o caso \emph{periódico}, não o caso único. Com $a_k\geq1$,
\[
\boxed{\;q_k\geq F_k,\quad\text{com igualdade em TODO }k
\text{ só na sucessão de uns}\;}
\]
--- o \textbf{ouro é o real mais lento que existe}, e por isso uma só taxa
limita todos. A família metálica não é o escopo da construção: é o seu caso
extremo.

\smallskip\noindent\emph{Medido:} \code{tests/geometria\_real.c} ($15{:}0$).
\end{abstract}

\section{As oito leis são uma base ortonormal}\label{sec:orto}

Sobre $\mathbb{F}_2$ há uma só forma bilinear simétrica não degenerada:
$\langle a,b\rangle=\operatorname{paridade}(a\wedge b)$.

\begin{teorema}\label{thm:orto}
Com $e_k=2^{k}$ reservado à Lei $k$, a matriz de Gram é a identidade:
$\langle e_i,e_j\rangle=\delta_{ij}$. Logo
\[
\boxed{\;b_i=\langle b,e_i\rangle\;}
\]
--- um AND e uma paridade, que é \emph{ler o bit} ---, e
$b=\sum_i\langle b,e_i\rangle e_i$ reconstrói. $\square$
\end{teorema}
\colunas{a base é ortonormal (\code{thm:ortonormal})}
        {Gram $=I$ nos $64$ pares; $2048$ leituras}
        {\code{geometria\_real.c} \S L1}

\begin{observacao}[oito leis não são uma metáfora para oito bits]
A base ortonormal das oito leis \emph{realiza} $\mathbb{F}_2^{8}$. Cada
operação da face binária de uma unidade é, portanto, uma operação
\textbf{coordenada} em oito posições --- e o byte deixa de ser analogia com o
ASCII: é a \textbf{realização finita da base universal}.
\end{observacao}

\section{E é a ortonormalidade que autoriza o bit a bit}\label{sec:bitabit}

Dizer que «oito passos enchem um byte» é uma afirmação sobre
\emph{codificação}, e ela não demonstra que a álgebra \emph{opere} bit a bit.
O que o demonstra é $G=I$: com Gram identidade a extracção da coordenada não
custa sistema nenhum. Numa base qualquer seria preciso \textbf{inverter} a
Gram --- e o preço mede-se.

\begin{teorema}[o gume da base torcida]\label{thm:torcida}
Com $f_k=e_k\oplus e_{k+1}$, a Gram sai da identidade em $24$ dos $64$ pares, e
a leitura directa passa a dar a coordenada errada em \textbf{exactamente
metade} das $2048$ --- $1024$, que é o que o XOR de dois bits independentes dá.
\emph{O \code{uint8\_t} não é conveniência de máquina: é a base ortonormal
escrita em hardware.} $\square$
\end{teorema}

        {o custo da coordenada numa base não ortonormal}
        {\code{geometria\_real.c} \S L2}

\section{A torre $1\to2\to4\to8$, e o reuso}\label{sec:torre}

\begin{teorema}\label{thm:torre}
A dobra $T\oplus T^{*}$ duplica a dimensão e chega a oito em
\textbf{exactamente três} passos. E a passagem ao andar seguinte não cria
aritmética nova: \emph{duplica a estrutura pela dualidade}, pelo que a torre
$2,4,8,16,\ldots$ é a própria expansão da base operacional.

O que cresce é o \textbf{objecto}, não a máquina: um real precisa de infinitos
bits, e a base que os lê tem oito posições, \textbf{reutilizadas bloco a
bloco}. Medido num fluxo de $400$ decisões lido em $50$ blocos pelas mesmas oito
coordenadas, sem a base ganhar uma posição. $\square$
\end{teorema}
\colunas{a dobra $1\to2\to4\to8$ e o ciclo de período oito}
        {os oito bits reusados em cada andar}
        {\code{geometria\_real.c} \S L3}

\section{O dual produz a recorrência}\label{sec:dual}

\begin{teorema}\label{thm:dual}
Com $A_m=\begin{psmallmatrix}m&1\\1&0\end{psmallmatrix}$ e $\sigma$ a classe de
$x$ em $\mathbb{Z}[x]/(x^{2}-mx-1)$:
\[
\sigma\dg=-\sigma^{-1}=m-\sigma,
\qquad \sigma+\sigma\dg=m,
\qquad \sigma\sigma\dg=-1,
\]
por \textbf{três caminhos independentes} --- o traço e o determinante da
companheira; a involução lida \emph{dentro} de $\mathbb{Z}[\sigma]$ usando só
$\sigma^{2}=m\sigma+1$; e a potência
$A_m^{k}=\begin{psmallmatrix}F_{k+1}&F_k\\F_k&F_{k-1}\end{psmallmatrix}$.
Concordam para $m=1,\ldots,40$, e \emph{nenhuma raiz é avaliada}: a raiz é a
representação alcançada, a recorrência é a construção. \emph{Gume}: mexida uma
entrada da companheira, o determinante deixa de valer $-1$ nos $40$. $\square$
\end{teorema}
\colunas{o dual, e o Gato como Möbius}
        {a recorrência inteira $u_{n+2}=m\,u_{n+1}+u_n$}
        {\code{geometria\_real.c} \S L4; \code{metalica.c}}

\section{Pisot: o operador produz o inteiro}\label{sec:pisot}

\begin{teorema}\label{thm:pisot}
$t_k=\sigma^{k}+(\sigma\dg)^{k}\in\mathbb{Z}$, logo
$\sigma^{k}=t_k-(\sigma\dg)^{k}$ e
\[
\boxed{\;\operatorname{dist}(\sigma^{k},\mathbb{Z})=\abs{\sigma\dg}^{k}\;}
\]
a partir do momento em que a outra face entrou na metade certa do intervalo.
A condição decide-se \textbf{em inteiros}, por
$(t_k-1)^{2}<\Delta F_k^{2}<(t_k+1)^{2}$ com $\Delta=m^{2}+4$, e concorda com a
forma reduzida pela identidade $t_k^{2}-\Delta F_k^{2}=4(-1)^{k}$ nos $131$
andares, sem uma discordância.

\emph{Não há escolha de aproximação: o inteiro $t_k$ é produzido pelo próprio
operador.} $\square$
\end{teorema}
\colunas{$\abs{\sigma}>1$, $\abs{\sigma\dg}<1$, vazamento zero (\code{thm:pisot})}
        {a distância ao inteiro como quantidade PRODUZIDA}
        {\code{geometria\_real.c} \S L5; \code{pisotbase.c} \S B4}

\begin{observacao}[o escopo, medido]\label{obs:escopo}
A condição $\abs{\sigma\dg}^{k}<\tfrac12$ quebra em \textbf{exactamente um}
andar de toda a janela: o ouro no primeiro degrau, onde
$\abs{\varphi\dg}=0{,}618>\tfrac12$ e o inteiro mais próximo de $\varphi$ é $2$,
não $1$. É o andar em que a outra face ainda não entrou na metade certa. Entra
no segundo degrau, e para $m\geq2$ já no primeiro.
\end{observacao}

\section{Os dois determinantes --- e a igualdade é o teorema}\label{sec:det}

Aqui há uma distinção que a primeira versão deste paper misturou, e que é
precisa:

\begin{teorema}\label{thm:doisdet}
São \textbf{dois objectos}, calculados por caminhos que não se tocam:
\[
\underbrace{\det A_m^{k}=F_{k+1}F_{k-1}-F_k^{2}}_{\text{das ENTRADAS, em }\mathbb{Z}}
\qquad\text{e}\qquad
\underbrace{\sigma^{k}(\sigma\dg)^{k}=N(\sigma^{k})=a^{2}+mab-b^{2}}_{\text{ESPECTRAL, o produto das duas faces}}
\]
O primeiro não sabe que existem valores próprios; o segundo não olha para uma
entrada da matriz. Coincidem --- ambos $(-1)^{k}$, nos $115$ andares, e junto
com a multiplicatividade $N(\sigma)^{k}=(-1)^{k}$ --- por «$\det$ é o produto do
espectro», \emph{que é o teorema}. É a coincidência que se mede, não a
definição. $\square$
\end{teorema}
\colunas{$\det$ É a medida (Teorema do Gato)}
        {as duas leituras da mesma unidade, separadas}
        {\code{geometria\_real.c} \S L6}

\begin{observacao}[e é daí que vem a geometria]\label{obs:area}
Nas duas direcções próprias, uma cresce e a outra encolhe:
$\abs{\sigma}>1$, $\abs{\sigma\dg}<1$, com
$\abs{\sigma^{k}(\sigma\dg)^{k}}=1$. \textbf{O intervalo não é postulado
pequeno; ele fica pequeno porque uma face cresce enquanto a face dual é
obrigada a encolher para preservar a unidade.} E «área», «comprimento» e
«largura» nomeiam aqui um determinante, que numa matriz de inteiros é um
inteiro: nenhum contínuo foi pedido emprestado para dizer isto.
\end{observacao}

\section{O encaixe}\label{sec:encaixe}

\begin{teorema}\label{thm:encaixe}
Os convergentes caem alternadamente nos dois lados, cada intervalo está
\textbf{dentro} do anterior, e por Cassini metálico
$F_{k+1}^{2}-F_{k+2}F_k=(-1)^{k}$ a largura é
\[
\abs{c_{k+1}-c_k}=\frac{1}{F_kF_{k+1}},
\]
exacta e sem uma divisão. $\square$
\end{teorema}

\section{$\mathbb{R}$ inteiro --- e o ouro é o extremo}\label{sec:todos}

Esta é a secção que a versão anterior não tinha, e a sua falta era uma
ressalva inventada pelo autor.

\begin{definicao}\label{def:real}
Um real \emph{é} uma sucessão de quocientes parciais $a_k\geq1$, com
convergentes $p_k/q_k$ dados por $q_{-1}=0$, $q_0=1$,
$q_k=a_kq_{k-1}+q_{k-2}$. A bijecção entre reais e essas sucessões é clássica e
cita-se.
\end{definicao}

\begin{teorema}[a construção corre sobre quocientes arbitrários]\label{thm:todos}
Para uma sucessão \textbf{qualquer} --- não só a periódica ---:
$p_kq_{k-1}-p_{k-1}q_k=(-1)^{k-1}$, logo $\abs{\det}=1$ em todos os passos; os
convergentes alternam de lado; os intervalos aninham; e $q_k$ cresce. Medido em
$220$ sucessões (o ouro, a prata e $218$ arbitrárias), $3651$ pares, sem
excepção. \emph{O metálico é o caso periódico, e não o caso único.} $\square$
\end{teorema}

\begin{teorema}[o ouro é o real mais lento que existe]\label{thm:ouro}
Com $a_k\geq1$,
\[
\boxed{\;q_k\geq F_k\ \text{para todo real},\qquad
\text{com igualdade em TODO }k\ \text{só na sucessão de uns.}\;}
\]
Logo a largura $1/(q_kq_{k+1})$ de \emph{qualquer} real é menor ou igual à do
ouro: \textbf{uma só taxa limita $\mathbb{R}$ inteiro}, e é a do ouro. A família
metálica não é o escopo da construção --- é o seu caso \textbf{extremo}.
$\square$
\end{teorema}
\colunas{a torre, o dual, e a família metálica}
        {a construção para TODOS os reais, com o ouro a limitar}
        {\code{geometria\_real.c} \S L8}

\begin{observacao}[o empate pontual é comum; o total é único]
A primeira redacção afirmou de mais. O empate $q_k=F_k$ num $k$ isolado é
frequente --- qualquer sucessão que comece por uns empata no princípio, e a
medição contou $69$. O que \emph{só} o ouro faz é empatar em \textbf{todos} os
andares: uma sucessão em $220$, e nenhuma fora da sucessão de uns. A afirmação
certa é essa, e a medição é que a fixou.
\end{observacao}

\section{O corte, e o bit que decide}\label{sec:corte}

O corte não é uma técnica adicional: é a \textbf{leitura ordenada da mesma
sucessão}.

\begin{teorema}\label{thm:corte}
Para $a\in\mathbb{Z}$, $b\geq1$ e $\Delta=m^{2}+4$, a pergunta
$\tfrac ab<\sigma_m$ converte-se na comparação inteira
\[
2a-mb<0\quad\text{ou}\quad(2a-mb)^{2}<b^{2}\Delta .
\]
Nos $19\,360$ racionais varridos: nenhum cai \emph{em cima} --- porque $\Delta$
não é quadrado, que é dizer $\sigma\notin\mathbb{Q}$ --- e nenhuma das duas
classes tem extremo, com o seguinte \emph{exibido} em cada caso. $\square$
\end{teorema}

\noindent Portanto o bit não apenas \emph{representa} o número:

\begin{corolario}\label{cor:bitdecide}
\[
\boxed{\;\text{bit}\to\text{decisão}\to\text{intervalo}\to\text{encaixe}
\to\text{corte}\to\text{número real}.\;}
\]
Cada bisseção é \emph{uma} pergunta do Teor.~\ref{thm:corte} e devolve
\textbf{um bit}; oito enchem um byte, que é o bloco que a base ortonormal do
\S\ref{sec:orto} lê pelas suas oito coordenadas. A base das oito leis reaparece
como \textbf{a máquina de leitura de cada bloco de decisões}. $\square$
\end{corolario}

\section{A completude}\label{sec:completude}

O intervalo aperta porque uma face é obrigada a encolher para a unidade se
conservar (Obs.~\ref{obs:area}); o encaixe fecha; o corte fica determinado.
As hipóteses de corpo ordenado completo estão medidas --- corpo, ordem
compatível com as operações, arquimediano com o $n$ exibido, completo pelo
corte --- em \code{tests/ordenado.c} ($7{:}0$).

\begin{observacao}[a identificação com $\mathbb{R}$]
Que um corpo ordenado completo seja isomorfo a $\mathbb{R}$, e único a menos de
isomorfismo, é teorema clássico, e cita-se como tal. \textbf{Mas a passagem a
$\mathbb{R}$ não é a identificação posterior de um subconjunto conveniente}: é
a realização da \emph{face alcançada} da própria construção. A face binária
permanece finita e não ordenável como corpo; a face alcançada é aquela em que
os cortes se tornam estrutura ordenada e completa. \emph{São duas faces do
mesmo eixo, não duas teorias concorrentes.}
\end{observacao}

\section{E só então $\pi$}\label{sec:pi}

\begin{teorema}\label{thm:pi}
Pela tricotomia do traço: os polígonos são os membros \textbf{elíticos}, e a
companheira $C=\begin{psmallmatrix}t&-1\\1&0\end{psmallmatrix}$ tem ordem
exactamente $2n$ --- $6$ no triângulo, $8$ no quadrado, $12$ no hexágono ---
com as raízes \emph{realizadas inteiras} nos primos das ordens: $\sqrt2$ é $11$
e $6$ em $\mathbb{F}_{17}$, $\sqrt3$ é $9$ e $4$ em $\mathbb{F}_{13}$. Os
metais são os \textbf{hiperbólicos}, que foram o motor
dos \S\S\ref{sec:dual}--\ref{sec:todos}. E o círculo é o \textbf{parabólico}
$t=2$, cuja ordem módulo $p$ é exactamente $p$ --- ímpar, logo nunca divisor da
ordem $2n$ de andar nenhum. $\square$
\end{teorema}

\[
\boxed{\;\pi_k=\text{fecho geométrico do andar }k,
\qquad \pi_\infty=\text{fronteira parabólica.}\;}
\]

\noindent Nenhuma constante foi importada: \textbf{cada andar apresenta a sua}.
E ao chegar ao círculo a família muda de natureza, de fechar na escada para
fechar só no primo --- é a mesma fronteira do eixo, agora como propriedade do
membro-limite. \emph{Assim o círculo não é usado para construir a reta: a reta
construída explica por que o círculo aparece como o seu gume.}

\section{O contraexemplo ao mecanismo}\label{sec:contra}

\begin{proposicao}\label{prop:contra}
Em $x^{2}-2x-4$ a recorrência é inteira e o traço é inteiro nos $12$ andares
--- \emph{as primeiras setas passam} --- e mesmo assim o critério do
\S\ref{sec:pisot} falha nos $12$ e $\abs{\det}=1$ em nenhum, porque
$\det A^{k}=(-4)^{k}$ \textbf{cresce} em módulo a cada andar. A face que devia
encolher não encolhe. $\square$
\end{proposicao}

\noindent Daqui não se conclui caracterização nenhuma --- «tem de ser unidade e
Pisot» não está demonstrado neste quadro. Conclui-se o que se vê: \textbf{em que
seta este exemplo pára}. É um contraexemplo ao \emph{mecanismo}, e serve para
mostrar que o mecanismo tem conteúdo.

\section{A construção está fechada}\label{sec:fechada}

O \emph{Corpo Universal} fornece a estrutura operacional; este paper fornece a
sua realização geométrica.

\begin{center}
\fbox{\parbox{0.94\textwidth}{\centering\medskip
o \textbf{bit} fornece a unidade\quad$\to$\quad
as \textbf{oito leis} fornecem a base\quad$\to$\quad
o \textbf{dual} fornece a recorrência\\[3pt]
$\to$\quad \textbf{Pisot} fornece o encolhimento\quad$\to$\quad
o \textbf{corte} fornece a ordem\quad$\to$\quad$\mathbb{R}$.\medskip}}
\end{center}

\medskip
\noindent E a tese, numa frase: \emph{a geometria real é a realização alcançada
de uma estrutura binária de oito leis --- a base fornece as coordenadas, a
dualidade fornece a recorrência, a família de Pisot fornece o encaixe, e o corte
fornece o contínuo.} Não usámos a reta para construir os números: construímos a
reta a partir do processo que os produz.
